-- 飞行物命中
function ProcessFlyerAttack(battle, flyer, skillId, targetEntry)
    local releaseEntry = flyer.releaseEntry
    -- 获取飞行物技能
    local context = skillContext.CreateSkillContext(battle, releaseEntry, skillId)
    table.insert(context.targetEntries, targetEntry)

    -- 释放技能
    releaseSkill(context)
end

---@param battle Battle
---@param sutra Sutra
function ProcessSutraAttack(battle, sutra, nowMS)
    local skillId = sutra:GetActiveSkill()
    if not skillId or skillId == 0 then return end

    if sutra:CanUseSkill(skillId) then
        local skillConfig = _G.SkillTable[skillId]
        if not skillConfig then
            LogErr("ProcessSutraAttack, cannot find skill config in table, sutraId:{} skillId:{}", sutra.sutraId, skillId)
            return
        end
        local context = skillContext.CreateSkillContext(battle, sutra, skillId)

        -- 释放技能
        releaseSkill(context)
    end
end

-- 被动释放技能
function ProcessEntrySkill(battle, entry, skillId)
    local skillConfig = _G.SkillTable[skillId]
    if not skillConfig then return end

    if entry:CanUseSkill(skillId) then
        -- 释放自动释放的技能
        local context = skillContext.CreateSkillContext(battle, entry, skillId)
        context.isTriggerUseSkillEvent = false

        -- 释放技能
        releaseSkill(context)
    end
end

-- 取消所有范围效果类型技能
function ProcessEntryCancelPassiveSkill(battle, entry, skillId)
    ---@type SkillTableData
    local skillConfig = _G.SkillTable[skillId]
    if not skillConfig then
        LogErr("ProcessFormationCancelPassiveSkill, cannot find skill config in table, skillId:{}", skillId)
        return
    end
    local battleTeam = battle.battleTeams[entry.teamId]

    for _, addAreaEffect in pairs(skillConfig.addAreaEffects) do
        local rangeEffectConfig = _G.AreaEffectTable[addAreaEffect]
        if not rangeEffectConfig then
            LogErr("ProcessFormationCancelPassiveSkill, cannot find range effect config in table, skillId:{} addAreaEffect:{}", skillId, addAreaEffect)
            goto process_entry_cancel_passive_skill_continue
        end

        if rangeEffectConfig.targetTeamType == 1 then
            -- 己方阵营
            battle:RemoveRangeEffect(battleTeam.teamId, addAreaEffect, battleTeam.tower.entryId)
        elseif rangeEffectConfig.targetTeamType == 2 then
            -- 敌方阵容
            for teamId, team in pairs(battle.battleTeams) do
                if teamId ~= battleTeam.teamId then
                    battle:RemoveRangeEffect(teamId, addAreaEffect, team.tower.entryId)
                end
            end
        end

        ::process_entry_cancel_passive_skill_continue::
    end
end

-- 范围效果 对单体释放技能
function ProcessRangeEffectSkill(battle, rangeEffect, skillId, entry)
    -- 如果范围技能不是由实体释放的，释放者就为范围技能本身
    local releaseEntry = rangeEffect.releaseEntry
    if not releaseEntry then
        releaseEntry = rangeEffect
    end

    local context = skillContext.CreateSkillContext(battle, releaseEntry, skillId)
    table.insert(context.targetEntries, entry)
    context.isTriggerUseSkillEvent = false

    releaseSkill(context)
end

-- 范围效果 对多实体释放技能
function ProcessRangeEffectSkillToEntries(battle, rangeEffect, skillId, entries)
    -- 如果范围技能不是由实体释放的，释放者就为范围技能本身
    local releaseEntry = rangeEffect.releaseEntry
    if not releaseEntry then
        releaseEntry = rangeEffect
    end

    local context = skillContext.CreateSkillContext(battle, releaseEntry, skillId)
    for _, entry in pairs(entries) do
        table.insert(context.targetEntries, entry)
    end
    context.isTriggerUseSkillEvent = false

    releaseSkill(context)
end

-- 释放符箓
---@param battle Battle
function ProcessSpellActiveSkill(battle, player, card, nowMS, count)
    -- 获取法术牌技能
    local config = _G.SpellTable[card.id]
    if not config then
        LogErr("ProcessSpellActiveSkill Spell config is nil, cardId:{}", card.id)
        return errCode.BATTLE_ERROR_SERVER_CFG
    end

    local skillId    = config.spellSkill[card.level]
    local battleTeam = battle.battleTeams[player.teamId]
    local tower      = battleTeam.tower

    -- 释放主动释放的技能
    local context = skillContext.CreateSkillContext(battle, tower, skillId)
    context.skillCoolingDown = battleTeam:GetSpellCD()

    -- 释放技能
    for i=1, count do
        releaseSkill(context)
    end

    return errCode.SUCCESS
end

---@param monster Monster
function ProcessMonsterPassiveSkill(battle, monster)
    local config = monster.monsterConfig
    for _, skillId in pairs(config.passiveSkills) do
        if monster:CanUseSkill(skillId) then
            local skillConfig = _G.SkillTable[skillId]
            if not skillConfig then

            -- 释放等级
            releaseSkill(context)
        end
    end
end

-- 取消被动技能，如果是buff就取消buff
---@param battle Battle
---@param sutra Sutra
function ProcessSutraCancelPassiveSkill(battle, sutra, passiveSkillId)
    ---@type SkillTableData
    local skillConfig = _G.SkillTable[passiveSkillId]
    if not skillConfig then
        LogErr("ProcessSutraCancelPassiveSkill skill config is nil, skillId:{}", passiveSkillId)
        return
    end

    local battleTeam = battle.battleTeams[sutra.teamId]

    for _, buffId in pairs(skillConfig.addBuffs) do
        buffComp.RemoveBuff(battle, buffId, sutra)
    end

    for _, addAreaEffect in pairs(skillConfig.addAreaEffects) do
        local rangeEffectConfig = _G.AreaEffectTable[addAreaEffect]
        if rangeEffectConfig.targetTeamType == 1 then
            -- 己方阵营
            battle:RemoveRangeEffect(battleTeam.teamId, addAreaEffect, sutra.entryId)
        elseif rangeEffectConfig.targetTeamType == 2 then
            -- 敌方阵容
            for teamId, _ in pairs(battle.battleTeams) do
                if teamId ~= battleTeam.teamId then
                    battle:RemoveRangeEffect(teamId, addAreaEffect, sutra.entryId)
                end
            end
        end
    end
end

---@param context SkillContext
function releaseSkill(context)
    if not context then
        return false
    end

    -- 判断技能cd
    local battle       = context.battle
    local skillId      = context.skillId
    local skillConfig  = context.skillConfig
    local releaseEntry = context.releaseEntry

    -- 选择目标阵营
    skillTarget.UpdateSkillTargetTeam(context)

    if not context.targetTeam then
        return true
    end

    if not next(context.targetEntries) then
        -- 选择作用目标
        local targetCount = skillTarget.UpdateSkillTarget(context)
        if targetCount <= 0 then
            return false
        end
    end

    -- 技能概率
    local probability = skillConfig.probability + releaseEntry:GetAttribute(_G.AttributeType.SKILL_EFFECTIVE_PROBABILITY)
    local random      = math.random(1, 10000)
    if random <= probability then
        -- 如果有前摇，则修改状态
        if context.skillConfig.preActionTime > 0 then
            statusMgr.ChangeStatus(releaseEntry, _G.EntryStatus.ENTRY_STATUS_ATTACK, context.skillConfig.preActionTime)

            -- 前摇结束后再释放技能
            local timerId = battle.timer:AddTimer(nil, context.skillConfig.preActionTime / 1000, releaseSkillImpl, context)
            releaseEntry.skillTimerId = timerId
        else
            releaseSkillImpl(context)
        end
    end

    -- 更新技能cd
    local nextSkillTime = releaseEntry.skillComponent:UpdateSkillCD(skillId, context.skillCoolingDown)

    battleIntent.InsertBattleIntent(battle.battleId, _G.INTENT_TYPE.USE_SKILL, {
        teamId  = releaseEntry.teamId,
        entryId = releaseEntry.entryId,
        skillId = skillId,
        targetInfo = getSkillContextTargetInfo(context),
        nextSkillTime = nextSkillTime
    });
    return true;
end

function releaseSkillImpl(context)
    local battle       = context.battle
    local releaseEntry = context.releaseEntry

    -- 删除前摇回调
    releaseEntry.skillTimerId = nil
    if context.isTriggerUseSkillEvent then
        -- 触发技能前事件
        local eventBeforeAttack = eventManager.CreateBeforeUseSkillEvent(releaseEntry, context)
        eventManager.DispatchEvent(battle.eventManager, eventBeforeAttack)
    end

    -- 技能生效
    if context.skillConfig.effectDelay > 0 then
        battle.timer:AddTimer(nil, context.skillConfig.effectDelay / 1000, SkillEffectExec, context)
    else
        SkillEffectExec(context)
    end

    if context.isTriggerUseSkillEvent then
        -- 触发技能后事件
        local eventAfterAttack = eventManager.CreateAfterUseSkillEvent(releaseEntry, context)
        eventManager.DispatchEvent(battle.eventManager, eventAfterAttack)
    end
end

---@param context SkillContext
function SkillEffectExec(context)
    local battle = context.battle
    if not battle or battle.status == _G.BattleStatus.END then
        return
    end

    for _, targetEntry in pairs(context.targetEntries) do
        -- 判断技能生效是否通过飞行物
        if context.skillConfig.byFlyer then
            local flyer = battle:CreateFlyer(context.targetTeam.teamId, context.skillConfig.flyerId,
                context.releaseEntry, targetEntry)
            if not flyer then
                goto continue
            end

            local distance = grid.GetDistance(context.releaseEntry.position, targetEntry.position)
            local speed = flyer:GetAttribute(_G.AttributeType.FINAL_MOVE_SPEED)

            battle.timer:AddTimer(nil, distance / speed, SkillEffectFlyerHit, flyer, context, targetEntry);
        else
            SkillEffectToTarget(context, targetEntry, 10000)
        end

        ::continue::
    end

    local skillConfig = context.skillConfig
    if skillConfig.skillType == _G.SkillType.SKILL_TYPE_ATTACK then
        -- 攻击后事件
        local event = eventManager.CreateEntryAfterAttackEvent(context.releaseEntry, context)
        eventManager.DispatchEvent(context.battle.eventManager, event)
    end
end

function SkillEffectToTarget(context, targetEntry, damageRate)
    damageRate = damageRate or 10000

    -- 造成伤害
    skillEffectDamage.Execute(context, targetEntry, damageRate)

    -- 施加buff
    skillEffectAddBuff.Execute(context, targetEntry)

    -- 生成范围效果
    skillEffectRangeEffect.Execute(context, targetEntry)
end

---@param flyer Flyer
function SkillEffectFlyerHit(flyer, context, targetEntry)
    local battle = context.battle;
    --LogDebug("SkillEffectFlyerHit battleId={} flyerId={}", battle.battleId, flyer.flyerId)
    -- 目标已经死亡，则换一个目标
    if flyer:CheckTargetIsDead() then
        if battle:ChangeFlyerTarget(flyer) then
            CalFlyerEffectTime(flyer, context, targetEntry)
        end
        return
    end

    local flayerDamageRate = flyer:PopDamageRate()

    -- 计算伤害
    SkillEffectToTarget(context, targetEntry, flayerDamageRate)

    if battle:UpdateFlyerCatapultTarget(flyer) then
        CalFlyerEffectTime(flyer, context, targetEntry)
    end
end

function CalFlyerEffectTime(flyer, context, targetEntry)
    -- 换了目标
    local distance = grid.GetDistance(targetEntry.position, flyer.targetEntry.position)
    local speed = flyer:GetAttribute(_G.AttributeType.FINAL_MOVE_SPEED)

    if distance / speed < 0.1 then
        SkillEffectFlyerHit(flyer, context, flyer.targetEntry)
    else
        context.battle.timer:AddTimer(nil, distance / speed, SkillEffectFlyerHit, flyer, context, flyer.targetEntry);
    end
end

function getSkillContextTargetInfo(context)
    local skillTargetInfo = {
        targetEntries = {}
    }

    for _, targetEntry in pairs(context.targetEntries) do
        table.insert(skillTargetInfo.targetEntries, targetEntry:GetEntryData())
    end
    return skillTargetInfo
end

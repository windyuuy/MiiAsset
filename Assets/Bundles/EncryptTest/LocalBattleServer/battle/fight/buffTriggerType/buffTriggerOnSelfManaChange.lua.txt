local buffComponent = import("battle/components/buffComponent.lua")
local eventManager = import("battle/event/eventManager.lua")

function buffTrigger(component, battle, buff, triggerTypeIndex)
    local callback = function(listener, properties, event)
        local player = event.player

        if player.id ~= properties.playerId then
            return false
        end

        buffComponent.buffEffect(properties.buffComponent, properties.buff)
        return true
    end
    local properties = { playerId = buff.owner.playerId, buff = buff, buffComponent = component }
    local listenerId = eventManager.AddEventListener(battle.eventManager, _G.EventType.MANA_CHANGE, callback, properties, false)
    buff.listenerEventType = _G.EventType.MANA_CHANGE
    table.insert(buff.listenerIds, listenerId)
end

function InsertBattleIntent(battleId, intentType, dataMap)
    local info = GetBattleIntent(battleId);
    info["allIntentList"] = info["allIntentList"] or {};
    local allIntentList = info["allIntentList"];

    local dataName = intentType2DataName[intentType];
    if not dataName then return; end

    table.insert(allIntentList, #allIntentList+1, {
        intentType = intentType,
        [dataName] = dataMap
    });
    -- LogInfo("InsertBattleIntent intentType={}", intentType);
end

function InsertUserIntent(battleId, uid, intentType, dataMap)
    local info = GetBattleIntent(battleId);
    if intentType == _G.INTENT_TYPE.SETTLE then
        info.hasSettleIntent = true;
    end
    local playersIntentList = info.playersIntentList;
    playersIntentList[uid] = playersIntentList[uid] or {}
    local playerIntentList = playersIntentList[uid];

    local dataName = intentType2DataName[intentType];
    if not dataName then return; end
    table.insert(playerIntentList, {
        intentType = intentType,
        [dataName] = dataMap
    });
end

function ClearBattleIntent(battleId)
    battlesIntent[battleId] = {
        frameId = 0,
        allIntentList     = {},
        playersIntentList = {}
    }
end

function GetBattleIntent(battleId)
    battlesIntent[battleId] = battlesIntent[battleId] or {
        frameId = 0,
        allIntentList     = {},
        playersIntentList = {}
    };
    return battlesIntent[battleId];
end

function SendIntents(nowMS, battleId, battle)
    if battle.status < _G.BattleStatus.PLAY then
        return;
    end

    local intentInfo = GetBattleIntent(battleId);
    intentInfo.frameId = intentInfo.frameId + 1;
    intentInfo.hasSettleIntent = nil;

    -- 发送前进行清理
    battlesIntent[battleId] = {
        frameId = intentInfo.frameId,
        allIntentList     = {},
        playersIntentList = {}
    };

    battle.tickState = 100
    for id, player in pairs(battle.players) do
        local updateInfo = {
            battleId          = battleId,
            frameId           = intentInfo.frameId,
            occurTime         = nowMS,
            allIntentList     = intentInfo.allIntentList,
            playersIntentList = (intentInfo.playersIntentList or {})[player.id] or {}
        }

        if robotUtil.IsRobotUid(player.id) then
            battle.tickState = 101
            battle.lastIntents = updateInfo
            s2r.BattleLogic_NotifyBattleUpdate(player, updateInfo)
            battle.tickState = 102
            goto continue;
        end

        local ss = _G.GetAgentByUid(player.id);
        if not ss then
            goto continue;
        end
        if not ss.battleId or ss.battleId ~= battleId then
            goto continue;
        end

        if battle.robotStop and robotUtil.IsRobotUid(player.id) then
            goto continue;
        end

        -- 输出intentType
        --if updateInfo.allIntentList then
        --    for _, intent in pairs(updateInfo.allIntentList) do
        --        if intent.intentType == _G.INTENT_TYPE.ADD_BUFF then
        --            LogInfo("send intent, intent:{}", intent)
        --        end
        --    end
        --end
        battle.tickState = 103
        battle.lastIntents = updateInfo
        ss.Call("BattleLogic_NotifyBattleUpdate", updateInfo);
        battle.tickState = 104
        ::continue::
    end
end

-- 同步轮次信息给客户端
function NotifyRoundInfo(battle)
    local roundInfo = {
        roundNumber     = battle.roundNumber,
        roundEndTime    = battle.roundEndTime,
        monsterGroupId  = battle.currentRoundMonsterGroupId,
    }

    -- 同步轮次信息
    InsertBattleIntent(battle.battleId, _G.INTENT_TYPE.ROUND_INFO, roundInfo);
end

---@param battle Battle
function NotifyBattleStart(battle, nowMS)
    LogInfo("NotifyBattleStart, battleId={}", battle.battleId)
    local msg = {
        battleId = battle.battleId,
        occurTime = nowMS,
        teams = {}
    }

    for _, battleTeam in pairs(battle.battleTeams) do
        local data = battleTeam:GetBattleTeamData(battle)
        table.insert(msg.teams, data)
    end

    -- 同步战斗开始信息
    for _, player in pairs(battle.players) do
        if robotUtil.IsRobotUid(player.id) then
            s2r.BattleLogic_NotifyBattleStart(player, msg)
            goto continue;
        end

        local ss = _G.GetAgentByUid(player.id);
        if not ss then
            LogInfo("NotifyBattleStart, player not online, playerId={}", player.id)
            goto continue;
        end

        if not ss.battleId or ss.battleId ~= battle.battleId then
            goto continue;
        end

        LogInfo("NotifyBattleStart, modeId={} msg={} playerId:{}", battle.mode, msg, player.id)
        ss.Call("BattleLogic_NotifyBattleStart", msg);

        -- 下面是测试用的，可以删掉
        -- ss.Call("BattleLogic_BattleReadyResponse", {battleId=battle.battleId});

        _G.LogPlayer(ss, "StartBattle", {
            battleId = battle.battleId,
        });

        ::continue::
    end
end

function NotifyBattleStop(nowMS, battleId, battle)
    -- stop前需要先把intent的发送出去
    local info = GetBattleIntent(battleId);
    if info.hasSettleIntent then
        SendIntents(nowMS, battleId, battle);
    end

    -- 发送stop的消息
    for _, player in pairs(battle.players) do
        if robotUtil.IsRobotUid(player.id) then
            s2r.BattleLogic_NotifyBattleStop(player, {battleId=battleId, stopReason=battle.stopReason})
            goto continue;
        end

        if not player.svr then
            LogInfo("理论上只有测试的时候才会出现该问题");
            goto continue;
        end

        -- 通知修改玩家状态
        message.SendP2PMessage("lobby", player.svr, "BattleStop", player.id, battleId, battle.mode, battle.stopReason);

        local ss = _G.GetAgentByUid(player.id);
        if not ss then
            goto continue;
        end
        if not ss.battleId or ss.battleId ~= battleId then
            goto continue;
        end
        ss.Call("BattleLogic_NotifyBattleStop", {battleId=battleId, stopReason=battle.stopReason});

        _G.LogPlayer(ss, "StopBattle", {
            battleId = battleId,
            battle.stopReason
        });

        ::continue::
    end

    battlesIntent[battleId] = nil
end

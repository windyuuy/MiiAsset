---@param attributeComponent AttributeComponent
function GetAttribute(attributeComponent, attrId)
    if attributeComponent.attributeChangeFlags[attrId] then
        attributeComponent:UpdateAttribute(attrId)
    end

    local finalAttrFunc = _G.FinalAttrFunc[attrId]
    if finalAttrFunc then
        local ok, ret = xpcall(finalAttrFunc, debug.traceback, attributeComponent)
        if not ok then
            LogErr("GetAttribute failed, err:{} attributeId:{}", ret, attrId)
            return 0
        end

        return ret
    end

    local attr = _G.AttrShortName[attrId]
    if not attr then
        LogErr("GetAttribute failed, cannot find attr short name, attrId:{}", attrId)
        return 0
    end

    return attributeComponent.attributeBaseData[attr] + (attributeComponent.attributeDeltaData[attr] or 0)
end

-- 计算攻击力数值
function getAttack(attributeComponent)
    local baseAttack = GetAttribute(attributeComponent, _G.AttributeType.BASE_ATTACK)
    local extraAttack = GetAttribute(attributeComponent, _G.AttributeType.EXTRA_ATTACK)
    local perAttack = GetAttribute(attributeComponent, _G.AttributeType. PER_ATTACK)
    local levelPerAttack = GetAttribute(attributeComponent, _G.AttributeType.LEVEL_PER_ATTACK)
    local fixedAttack = GetAttribute(attributeComponent, _G.AttributeType.FIXED_ATTACK)

    --LogDebug("baseAttack:{} extraAttack:{} perAttack:{} levelPerAttack:{} fixedAttack:{}", baseAttack, extraAttack, perAttack, levelPerAttack, fixedAttack)
    return ((baseAttack * levelPerAttack / 100) + extraAttack) * (1 + perAttack / 100) + fixedAttack
end

-- 计算攻击速度数值
function getAttackSpeed(attributeComponent)
    local baseAttackSpeed = GetAttribute(attributeComponent, _G.AttributeType.BASE_ATTACK_SPEED)
    local perAttackSpeed = GetAttribute(attributeComponent, _G.AttributeType.PER_ATTACK_SPEED)

    return baseAttackSpeed / (1 + perAttackSpeed / 100)
end

-- 计算移动速度
function getMoveSpeed(attributeComponent)
    local baseMoveSpeed = GetAttribute(attributeComponent, _G.AttributeType.BASE_MOVE_SPEED)
    local perMoveSpeed = GetAttribute(attributeComponent, _G.AttributeType.PER_MOVE_SPEED)

    local ret = baseMoveSpeed * (1 + perMoveSpeed / 100)
    ret = math.max(50, ret)
    ret = math.min(300, ret)
    return ret
end

-- 计算最大血量数值
function getHealth(attributeComponent)
    local baseHealth = GetAttribute(attributeComponent, _G.AttributeType.BASE_HEALTH)
    local extraHealth = GetAttribute(attributeComponent, _G.AttributeType.EXTRA_HEALTH)
    local perHealth = GetAttribute(attributeComponent, _G.AttributeType.PER_HEALTH)
    local fixedHealth = GetAttribute(attributeComponent, _G.AttributeType.FIXED_HEALTH)

    local res = (baseHealth + extraHealth) * (1 + perHealth / 100) + fixedHealth
    return res
end

-- 计算暴击率数值
function getCritRate(attributeComponent)
    local baseCritRate = GetAttribute(attributeComponent, _G.AttributeType.BASE_CRIT_RATE)
    local extraCritRate = GetAttribute(attributeComponent, _G.AttributeType.EXTRA_CRIT_RATE)

    return baseCritRate + extraCritRate
end

-- 计算暴击伤害数值
function getCritDamage(attributeComponent)
    local baseCritDamage = GetAttribute(attributeComponent, _G.AttributeType.BASE_CRIT_DAMAGE)
    local extraCritDamage = GetAttribute(attributeComponent, _G.AttributeType.EXTRA_CRIT_DAMAGE)

    return (baseCritDamage + extraCritDamage) / 100
end

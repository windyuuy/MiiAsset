function GetOneRuinStatus(guildRuinId, guildRuinConfig, now)
    if guildRuinConfig.firstSignTime == 0 then
        return {
            status = _G.GuildRuinsStatus.GUILD_RUINS_STATUS_NO_OPEN
        }
    end

    if _G.GmStartGuildRuinId == guildRuinId and _G.GmStartGuildRuinData and _G.GmStartGuildRuinData.ruinDeadline > now then
        if now < _G.GmStartGuildRuinData.joinDeadline then
            status = _G.GuildRuinsStatus.GUILD_RUINS_STATUS_OPEN
        elseif now < _G.GmStartGuildRuinData.lockDeadline then
            status = _G.GuildRuinsStatus.GUILD_RUINS_STATUS_LOCK
        else
            status = _G.GuildRuinsStatus.GUILD_RUINS_STATUS_CLOSE
        end 
        return {
            startTime    = _G.GmStartGuildRuinData.startTime,
            joinDeadline = _G.GmStartGuildRuinData.joinDeadline,
            lockDeadline = _G.GmStartGuildRuinData.lockDeadline,
            ruinDeadline = _G.GmStartGuildRuinData.ruinDeadline,
            status       = status,
        }
    end

    local count = math.max(0, math.floor((now - guildRuinConfig.firstSignTime) / guildRuinConfig.openCycle))
    local startTime = count * guildRuinConfig.openCycle + guildRuinConfig.firstSignTime
    local joinDeadline = count * guildRuinConfig.openCycle + guildRuinConfig.firstSignTime + guildRuinConfig.signStopTime

    local status
    if now < guildRuinConfig.firstSignTime then
        status = _G.GuildRuinsStatus.GUILD_RUINS_STATUS_CLOSE
    elseif now < joinDeadline then
        status = _G.GuildRuinsStatus.GUILD_RUINS_STATUS_OPEN
    elseif now - joinDeadline < guildRuinConfig.signLockTime then
        status = _G.GuildRuinsStatus.GUILD_RUINS_STATUS_LOCK
    else
        status = _G.GuildRuinsStatus.GUILD_RUINS_STATUS_CLOSE
    end
    return {
        startTime    = startTime,
        joinDeadline = joinDeadline,
        lockDeadline = joinDeadline + guildRuinConfig.signLockTime,
        ruinDeadline = joinDeadline + guildRuinConfig.signLockTime + guildRuinConfig.ruinOpenDuring,
        status       = status,
    }
end

-- 获取当前有哪些界墟开启报名中
function GetAllGuildRuinsStatusData()
    local now = _G.GetCurTime()
    local guildRuinsStatus = {}
    for guildRuinId, guildRuinConfig in pairs(_G.GuildRuinsTable) do
        guildRuinsStatus[guildRuinId] = GetOneRuinStatus(guildRuinId, guildRuinConfig, now)
    end
    return guildRuinsStatus
end

function GetGuildRuinsStatusData(guildRuinId)
    local guildRuinConfig = _G.GuildRuinsTable[guildRuinId]
    if not guildRuinConfig then
        LogErr("GetGuildRuinsStatusData guildRuinId={} not exist!", guildRuinId)
        return
    end
    return GetOneRuinStatus(guildRuinId, guildRuinConfig, _G.GetCurTime())
end

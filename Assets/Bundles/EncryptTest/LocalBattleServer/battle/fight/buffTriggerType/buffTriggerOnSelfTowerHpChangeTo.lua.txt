-- 32. 己方基地血量小于指定比例时触发

local buffComponent = import("battle/components/buffComponent.lua")
local eventManager = import("battle/event/eventManager.lua")

function buffTrigger(component, battle, buff, triggerTypeIndex)
    local callback = function(listener, properties, event)
        local tower = event.tower
        local oldHP = event.oldHP
        local newHP = event.newHP

        if tower.teamId ~= properties.teamId then
            return false
        end

        local oldRate = oldHP / tower.maxHP * 10000
        local newRate = newHP / tower.maxHP * 10000
        local targetRate = properties.targetRate

        local result = oldRate > targetRate and newRate <= targetRate
        if result then
            buffComponent.buffEffect(properties.buffComponent, properties.buff)
        end

        return result
    end
    local buffConfig = _G.BuffTable[buff.buffId]
    local triggerTypeData = buffConfig.triggerTypeDatas[triggerTypeIndex]
    local targetRate = triggerTypeData[1]
    local properties = { teamId = buff.owner.teamId, buff = buff, buffComponent = component, targetRate = targetRate }
    local listenerId = eventManager.AddEventListener(battle.eventManager, _G.EventType.TOWER_HP_CHANGE_TO, callback, properties, false)
    buff.listenerEventType = _G.EventType.TOWER_HP_CHANGE_TO
    table.insert(buff.listenerIds, listenerId)
end

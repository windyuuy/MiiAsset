function CreateEventManager()
    local eventManager = {
        listener = {},
        nextListenerId = 0,
    }
    return eventManager
end

-- 注册一个事件监听器
-- eventType: 事件类型
-- callback: 回调函数
-- com.zhiyou.projectf.properties:　监听器额外参数
-- once: 是否只触发一次
function AddEventListener(eventManager, eventType, callback, properties, once)
    --LogDebug("AddEventListener, eventType:{}", eventType)
    if not eventManager.listener[eventType] then
        eventManager.listener[eventType] = {}
    end

    local listenerId = getNextListenerId(eventManager)
    local listener = eventListener.CreateListener(eventType, listenerId, callback, properties, once)
    eventManager.listener[eventType][listenerId] = listener
    listener.listenerId = listenerId
    return listenerId
end

-- 移除一个事件监听器
function RemoveEventListener(eventManager, eventType, listenerId)
    if not eventManager.listener[eventType] then
        return
    end

    eventManager.listener[eventType][listenerId] = nil
end

-- 触发事件
function DispatchEvent(eventManager, event)
    local eventType = event.eventType
    local listeners = eventManager.listener[eventType]
    if not listeners then
        return
    end

    local removeListenerIds = {}
    --LogDebug("DispatchEvent eventType:{}", eventType)

    for i, listener in pairs(listeners) do
        if not listener then
            LogDebug("DispatchEvent listener is nil, eventType:{}", eventType)
            goto continue
        end
        local isTrigger = listener:callback(listener.properties, event)
        --if isTrigger then
        --    LogDebug("DispatchEvent event is trigger, eventType:{}", eventType)
        --end
        if isTrigger and listener.once then
            table.insert(removeListenerIds, i)
        end

        ::continue::
    end

    for _, id in pairs(removeListenerIds) do
        eventManager.listener[eventType][id] = nil
    end
end

---@param equipList table<number, SlotData>
function CreateSutraOnEvent(teamId, equipList)
    local sutraIds = {}
    for _, slotData in pairs(equipList) do
        if slotData and slotData.card then
            table.insert(sutraIds, slotData.card.id)
        end
    end

    local event = {
        teamId = teamId,
        eventType = _G.EventType.SUTRA_ON,
        sutraIds = sutraIds,
    }
    return event
end

-- p2p消息
local tools   = import("common/tools.lua");
local pbUtils = import("common/pbUtils.lua");

-- 发送点对点消息
function SendP2PMessage(serviceId, instanceId, msg, ...)
    if not instanceId then
        LogErr("failed send p2p message instance id not exists msg:{}", msg);
        return false;
    end

    if dive.isStandalone then
        return true;
    end

    local data = {msg=msg, svrid=_G.diveSDMgr.get_local_instance_id(), params={ ... }};

    local res = dive.RemoteInvoke(serviceId, instanceId, data);

    if res == "ok" then
        return true;
    elseif res == "instanceId not exists" then
        return false;
    else
        LogErr("failed send p2p message async call failed msg:{} res:{}", msg, res);
        return false;
    end
end

-- 处理接受的消息
function HandleP2PMessage(id, timestamp, sourceServiceId, sourceInstanceId, data)
    -- 准备分发消息
    local proc = s2s[data.msg];
    if not proc then
        LogErr("undefined s2s msg: {}", data.msg);
        return true;
    end

    local ok, err = xpcall(proc, debug.traceback, data.svrid, table.unpack(data.params));
    if not ok then
        LogErr("failed to call msg s2s.{} err:{}", data.msg, err);
    end
end

-- 处理cs推送的消息
function HandleProtoMsgFrmCs(msg, data)
    LogDebug("{}", data)
    local proc = c2s[msg];
    if not proc then
        LogErr("undefined c2s msg: {}", msg);
        return true;
    end
    local ss = {}
    ss.uid = dive.playerUid
    ss.Call = function(msg, data)
        SendProtoMsgToCs(msg, data);
    end

    local tData = _G.jsonDecode(data)
    local ok, err = xpcall(proc, debug.traceback, ss, tData);
    if not ok then
        LogErr("failed to call msg c2s.{}", msg);
        LogErr("{}", err);
    end
end

-- 发送消息到cs
function SendProtoMsgToCs(msg, data)
    data = pbUtils.FormatData(msg, data)
    local strData = _G.jsonEncode(data)
    dive.SendProtoMsgToCs(msg, strData)
end

-- 发送消息到机器人
function SendMsgToLocalRobot(msg, data, ss)
    if not msg then
        LogErr("nil msg from session {}", token);
        return;
    end
    local proc = _G.s2lb[msg];
    if not proc then
        --LogErr("undefined msg {} from session {}", msg, token);
        return;
    end
    local ok, err = xpcall(proc, debug.traceback, ss, data);
    if not ok then LogErr("{}", err); end
end

-- 处理机器人消息
function HandleMsgFromLocalRobot(msg, data)
    if not msg then
        LogErr("nil msg from session {}", token);
        return;
    end

    local proc = _G.c2s[msg];
    if not proc then
        LogErr("undefined msg {} from session {}", msg, token);
        return;
    end

    local ss = _G.GetSessionByUid(_G.LOCAL_ROBOT_UID)
    local ok, err = xpcall(proc, debug.traceback, ss, data);
    if not ok then
        LogErr("failed to call msg c2s.{}", msg);
        LogErr("{}", err);
    end
end

-- 增加一条离线消息
function InsertOffMsg(uid, offMsgType, ...)
    CheckDbLogicMgrCallBack()
    local ok, message = xpcall(dive.JsonEncode, debug.traceback, {offMsgType=offMsgType, params={ ... }});
    if not ok then
        LogErr("InsertOffMsg failed to encode data msg={}", message);
        return ok;
    end
    ok = _G.dbLogicMgr.InsertOffMsgAsync(uid, message);
    if ok ~= "ok" then
        LogErr("InsertOffMsg failed msg={}", ok);
    end
    return ok;
end

function CheckDbLogicMgrCallBack()
    -- 插入离线消息的回调
    if _G.dbLogicMgr and not _G.dbLogicMgr.InsertOffMsgCallback then
        _G.dbLogicMgr.InsertOffMsgCallback = tools.LogDecorator(function(result, uid, message)
            if not result then
                LogErr("InsertOffMsgCallback uid={} message={}, result:{}", uid, message, result);
            end
        end);
    end
end

local get_filenode = function(filename)
    local node = dive.files[filename];
    if node then
        return node;
    end

    local env = dive.CreateEnv();
    --setmetatable(env, dive.meta);
    node = {env=env, filename=filename, loaded=false};
    dive.files[filename] = node;
    return node;
end

local try_load = function(node)
    local body = dive.LoadLuaSourceFile(node.filename);
	if not body then
		PrintError("open file: {} ... ... [failed]", node.filename);
		return;
	end

    --remove BOM. loadfile() can skip BOM, but load() cannot
    if string.byte(body,1)==239 and string.byte(body,2)==187 and string.byte(body,3)==191 then
        print("load file: {} ... ... [with BOM]", node.filename);
        body=string.char(string.byte(body,4,string.len(body)))
    end

    --keep original line numbers, for xpcall etc.
    local header = string.gsub(dive.header_sys, "%c", " ")
    -- local header = dive.header_sys

    --remove comment. loadfile() can skip first line as comment if it starts with '#', but load() cannot
    if '#' == string.sub(body, 1, 1) then
        body = string.gsub(body, "#[^\n]*\n", header.."\n", 1)
    else
        body = header .. body;
    end

    node.loaded = true;

    local trunk, msg = load(body, "@"..node.filename, "bt", node.env);
    if not trunk then
        return
    else
        trunk()
    end

    return node.env
end

dive.Reload = function(filename)
	if not dive.files[filename] then
		return;
	end

	local node = get_filenode(filename);

	try_load(node);
end

function _G.import(filename)
	local node = get_filenode(filename);

	if node.loaded then
	   return node.env;
	end

	node.env = try_load(node);
    return node.env;
end

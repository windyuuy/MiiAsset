-- 23. 召唤怪物
local battleMgr = import("battle/battleMgr.lua")

-- buff生效
---@param buff Buff
function BuffEffect(buff)
    local battle      = battleMgr.GetBattle(buff.owner.battleId)
    local buffConfig  = _G.BuffTable[buff.buffId]
    local monsterId   = buffConfig.effectData[1]    -- 召唤怪物id
    local monsterCount = buffConfig.effectData[2]    -- 召唤怪物数量
    local campType    = buffConfig.effectData[3]    -- 阵营类型

    for i = 1, monsterCount do
        if campType == 1 then
            -- 己方
            ---@type BattleTeam
            local battleTeam = battle.battleTeams[buff.owner.teamId]
            for routeGroupId, _ in pairs(battleTeam.monsterPaths) do
                battle:PushMonsterQueue(buff.owner.teamId, routeGroupId, monsterId, false, false, battle.roundNumber)
            end
        elseif campType == 2 then
            -- 敌方
            for teamId, battleTeam in pairs(battle.battleTeams) do
                if teamId ~= buff.owner.teamId then
                    for routeGroupId, _ in pairs(battleTeam.monsterPaths) do
                        battle:PushMonsterQueue(teamId, routeGroupId, monsterId, false, false, battle.roundNumber)
                    end
                end
            end
        elseif campType == 3 then
            -- 双方
            for teamId, battleTeam in pairs(battle.battleTeams) do
                for routeGroupId, _ in pairs(battleTeam.monsterPaths) do
                    battle:PushMonsterQueue(teamId, routeGroupId, monsterId, false, false, battle.roundNumber)
                end
            end
        end
    end
end

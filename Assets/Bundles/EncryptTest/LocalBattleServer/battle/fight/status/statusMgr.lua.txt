function CheckStatusEnd(entry)
    local statusComp = entry.statusComponent
    if statusComp.status == statusComp.defaultStatus then
        return;
    end

    local nowMS = _G.GetCurTimeMS();

    -- 状态结束, 都变为IDLE状态
    if statusComp.duration > 0 and nowMS - statusComp.startTime > statusComp.duration then
        statusComp.status = statusComp.defaultStatus;
        statusComp.startTime = _G.GetCurTimeMS();
        statusComp.duration = 0;
        statusComp.statusHandler[statusComp.status](entry);
    end
end

function ChangeStatus(entry, newStatus, duration, param)
    local statusComp = entry.statusComponent
    if not statusComp then
        return
    end

    CheckStatusEnd(entry)

    if statusComp.defaultStatus == newStatus or canChangeStatus(statusComp.transitionTable, statusComp.status, newStatus) then
        statusComp.status = newStatus;
        statusComp.startTime = _G.GetCurTimeMS();
        statusComp.duration = duration or 0;
        statusComp.param = param

        statusComp.statusHandler[statusComp.status](entry);

        if entry.attributeComponent then
            return entry.attributeComponent:OnStatusChange(newStatus, param)
        end
    else
        LogInfo("transitionToStatus can not transition from status:{} to status:{}", statusComp.status, newStatus)
        return
    end
end

function CheckStatus(entry, newStatus)
    local nowMS = _G.GetCurTimeMS();
    local statusComp = entry.statusComponent

    CheckStatusEnd(entry)

    if statusComp.status == newStatus then
        return true;
    end

    if canChangeStatus(statusComp.transitionTable, statusComp.status, newStatus) then
        return true
    else
        return false
    end
end

function canChangeStatus(transitionTable, oldStatus, newStatus)
    for _, status in pairs(transitionTable[oldStatus]) do
        if status == newStatus then
            return true
        end
    end

    return false
end

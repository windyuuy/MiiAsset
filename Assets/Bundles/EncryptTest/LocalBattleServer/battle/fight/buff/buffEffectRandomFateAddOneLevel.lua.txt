-- 36. 随机羁绊计数+1

local battleMgr = import("battle/battleMgr.lua")

-- buff生效
function BuffEffect(buff)
    -- LogDebug("buffEffectRandomFateAddOneLevel:BuffEffect buffId={}", buff.buffId)
    local battle     = battleMgr.GetBattle(buff.owner.battleId)
    local battleTeam = battle.battleTeams[buff.owner.teamId]

    local isUpdate = battleTeam.fateComponent:RandomUpgradeFate(battle.battleId, battleTeam.teamId, buff)
    if isUpdate then
        battleTeam:UpdateFates()
    end
end

-- 取消buff
function UnBuffEffect(buff)
    local battle     = battleMgr.GetBattle(buff.owner.battleId)
    local battleTeam = battle.battleTeams[buff.owner.teamId]
    battleTeam.fateComponent:RemoveFateBuff(buff)
end

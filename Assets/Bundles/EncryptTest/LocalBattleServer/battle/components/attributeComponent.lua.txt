local battleIntent = import("battle/battleIntent.lua")
local reason = import("common/reason.lua")
local buffEffectAttribute = import("battle/fight/buff/buffEffectAttribute.lua")

---@class AttributeComponent 属性组件
---@field entry Entry 所属实体
---@field attributeBaseData table<string, number> 基础属性数据
---@field attributeDeltaData table<string, number> 局内附加属性数据
---@field attributeBuff table<number, table<number, Buff>> 影响属性的buff[属性id: [buffId: Buff]]
---@field attributeChangeFlags table<number, boolean> 是否改变属性
_G.AttributeComponent = class()

NOT_STATUS_SPEED = -99999

function CreateAttributeComponent(...)
    return AttributeComponent.new(...)
end

function AttributeComponent:ctor(entry)
    self.entry = entry
    self.attributeBaseData = {}
    self.attributeDeltaData = {}
    self.attributeBuff = {}
    self.attributeChangeFlags = {}

    self.statusMoveSpeed = NOT_STATUS_SPEED
end

function AttributeComponent:Init(config)
    self:InitBaseAttr(config)
end

function AttributeComponent:GetAttribute(attrId)
    return GetAttribute(self, attrId)
end

---@param buff Buff
function AttributeComponent:AddAttributeBuff(buff)
    local buffConfig = _G.BuffTable[buff.buffId]
    local effectData = buffConfig.effectData
    --LogDebug("AddAttributeBuff, buffId:{} effectData:{}", buff.buffId, effectData)
    local attrId = effectData[2]

    local oldHp = self:GetAttribute(_G.AttributeType.FINAL_HEALTH)

    self.attributeBuff[attrId] = self.attributeBuff[attrId] or {}
    self.attributeBuff[attrId][buff.buffId] = buff
    self.attributeChangeFlags[attrId] = true

    local newHp = self:GetAttribute(_G.AttributeType.FINAL_HEALTH)

    if oldHp ~= newHp then
        local oldCurHp = self.entry.curHP
        self.entry.curHP = newHp < oldHp and math.min(self.entry.curHP, newHp) or self.entry.curHP + (newHp - oldHp)

        battleIntent.InsertBattleIntent(self.entry.battleId, _G.INTENT_TYPE.ENTRY_ATTRIBUTE_CHANGE, {
            teamId = self.entry.teamId,
            entry = self.entry:GetEntryData(),
            attributeType = _G.ATTRIBUTE_TYPE.HEALTH,
            oldValue = oldCurHp,
            newValue = math.ceil(self.entry.curHP),
            reason = reason.BUFF_EFFECT_UPDATE_HP,
        })
    end
end

function AttributeComponent:RemoveAttributeBuff(buff)
    local buffConfig = _G.BuffTable[buff.buffId]
    local attrId = buffConfig.effectData[2]

    self.attributeBuff[attrId] = self.attributeBuff[attrId] or {}
    self.attributeBuff[attrId][buff.buffId] = nil
    self.attributeChangeFlags[attrId] = true
end

function AttributeComponent:UpdateAttribute(attrId)
    local attr = _G.AttrShortName[attrId]
    --local oldValue = self.attributeDeltaData[attr] or 0
    local value = 0
    for _, buff in pairs(self.attributeBuff[attrId]) do
        value = value + buffEffectAttribute.GetBuffValue(buff)
    end

    self.attributeDeltaData[attr] = value
    self.attributeChangeFlags[attrId] = false

    --LogDebug("UpdateAttribute, attr:{} attrId:{} old:{} new:{}", attr, attrId, oldValue, value)
    -- TODO: 需要变更时通知客户端的属性
    if attrId == _G.AttributeType.BASE_MOVE_SPEED or attrId == _G.AttributeType.EXTRA_MOVE_SPEED
        or attrId == _G.AttributeType.PER_MOVE_SPEED or attrId == _G.AttributeType.FIXED_MOVE_SPEED then
        -- 发送属性变化信息
        battleIntent.InsertBattleIntent(self.entry.battleId, _G.INTENT_TYPE.ENTRY_ATTRIBUTE_CHANGE, {
            teamId = self.entry.teamId,
            entry = self.entry:GetEntryData(),
            attributeType = _G.ATTRIBUTE_TYPE.MOVE_SPEED,
            newValue = getMoveSpeed(self),
            reason = reason.BUFF_UPDATE_MOVE_SPEED,
            position = self.entry.position,
            totalMoveDistance = self.entry.totalMoveDistance,
        })
    end
end

function AttributeComponent:OnStatusChange(status, param)
    local notifyClient = false;
    if status == _G.EntryStatus.ENTRY_STATUS_REPEL then
        if self.statusMoveSpeed ~= param.speed then
            self.statusMoveSpeed = param.speed
            notifyClient = true
        end
    elseif status == _G.EntryStatus.ENTRY_STATUS_DIZZY then
        if self.statusMoveSpeed ~= 0 then
            self.statusMoveSpeed = 0
            notifyClient = true
        end
    else
        if self.statusMoveSpeed ~= NOT_STATUS_SPEED then
            self.statusMoveSpeed = NOT_STATUS_SPEED
            notifyClient = true
        end
    end

    if notifyClient then
        -- 发送属性变化信息
        battleIntent.InsertBattleIntent(self.entry.battleId, _G.INTENT_TYPE.ENTRY_ATTRIBUTE_CHANGE, {
            teamId = self.entry.teamId,
            entry = self.entry:GetEntryData(),
            attributeType = _G.ATTRIBUTE_TYPE.MOVE_SPEED,
            newValue = getMoveSpeed(self),
            reason = reason.BUFF_UPDATE_MOVE_SPEED,
            position = self.entry.position,
            totalMoveDistance = self.entry.totalMoveDistance,
        })
    end
end

---@param attributeComponent AttributeComponent
function GetAttribute(attributeComponent, attrId)
    if attributeComponent.attributeChangeFlags[attrId] then
        attributeComponent:UpdateAttribute(attrId)
    end

    local finalAttrFunc = _G.FinalAttrFunc[attrId]
    if finalAttrFunc then
        local ok, ret = xpcall(finalAttrFunc, debug.traceback, attributeComponent)
        if not ok then
            LogErr("GetAttribute failed, err:{} attributeId:{}", ret, attrId)
            return 0
        end

        return ret
    end

    local attr = _G.AttrShortName[attrId]
    if not attr then
        LogErr("GetAttribute failed, cannot find attr short name, attrId:{}", attrId)
        return 0
    end

    return attributeComponent.attributeBaseData[attr] + (attributeComponent.attributeDeltaData[attr] or 0)
end

-- 计算攻击力数值
function getAttack(attributeComponent)
    local baseAttack = GetAttribute(attributeComponent, _G.AttributeType.BASE_ATTACK)
    local extraAttack = GetAttribute(attributeComponent, _G.AttributeType.EXTRA_ATTACK)
    local perAttack = GetAttribute(attributeComponent, _G.AttributeType.PER_ATTACK)
    local levelPerAttack = GetAttribute(attributeComponent, _G.AttributeType.LEVEL_PER_ATTACK)
    local fixedAttack = GetAttribute(attributeComponent, _G.AttributeType.FIXED_ATTACK)

    --LogDebug("baseAttack:{} extraAttack:{} perAttack:{} levelPerAttack:{} fixedAttack:{}", baseAttack, extraAttack, perAttack, levelPerAttack, fixedAttack)
    return ((baseAttack * levelPerAttack / 10000) + extraAttack) * (1 + perAttack / 10000) + fixedAttack
end

-- 计算攻击速度数值
function getAttackSpeed(attributeComponent)
    local baseAttackSpeed = GetAttribute(attributeComponent, _G.AttributeType.BASE_ATTACK_SPEED)
    local perAttackSpeed = GetAttribute(attributeComponent, _G.AttributeType.PER_ATTACK_SPEED)

    return baseAttackSpeed / (1 + perAttackSpeed / 10000)
end

-- 计算移动速度
function getMoveSpeed(attributeComponent)
    if attributeComponent.statusMoveSpeed ~= NOT_STATUS_SPEED then
        return attributeComponent.statusMoveSpeed
    end

    local baseMoveSpeed = GetAttribute(attributeComponent, _G.AttributeType.BASE_MOVE_SPEED)
    local perMoveSpeed = GetAttribute(attributeComponent, _G.AttributeType.PER_MOVE_SPEED)

    local ret = baseMoveSpeed * (1 + perMoveSpeed / 10000)
    ret = math.max(0, ret)
    ret = math.min(300, ret)
    return ret
end

-- 计算最大血量数值
function getHealth(attributeComponent)
    local baseHealth = GetAttribute(attributeComponent, _G.AttributeType.BASE_HEALTH)
    local extraHealth = GetAttribute(attributeComponent, _G.AttributeType.EXTRA_HEALTH)
    local perHealth = GetAttribute(attributeComponent, _G.AttributeType.PER_HEALTH)
    local fixedHealth = GetAttribute(attributeComponent, _G.AttributeType.FIXED_HEALTH)

    local res = (baseHealth + extraHealth) * (1 + perHealth / 10000) + fixedHealth
    return res
end

-- 计算暴击率数值
function getCritRate(attributeComponent)
    local baseCritRate = GetAttribute(attributeComponent, _G.AttributeType.BASE_CRIT_RATE)
    local extraCritRate = GetAttribute(attributeComponent, _G.AttributeType.EXTRA_CRIT_RATE)

    return baseCritRate + extraCritRate
end

-- 计算暴击伤害数值
function getCritDamage(attributeComponent)
    local baseCritDamage = GetAttribute(attributeComponent, _G.AttributeType.BASE_CRIT_DAMAGE)
    local extraCritDamage = GetAttribute(attributeComponent, _G.AttributeType.EXTRA_CRIT_DAMAGE)

    return (baseCritDamage + extraCritDamage)
end

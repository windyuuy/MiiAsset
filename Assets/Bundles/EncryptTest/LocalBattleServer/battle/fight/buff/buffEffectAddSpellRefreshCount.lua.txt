-- 17. 新增免费刷新次数
local battleMgr = import("battle/battleMgr.lua")
local gameCardComp = import("battle/components/gameCardComponent.lua")

-- buff生效
---@param buff Buff
function BuffEffect(buff)
    local buffConfig = _G.BuffTable[buff.buffId]

    local battle = battleMgr.GetBattle(buff.owner.battleId)
    local freeCount = buffConfig.effectData[1]

    for _, player in pairs(battle.players) do
        if player.teamId == buff.owner.teamId then
            gameCardComp.ChangeSpellRefreshCount(player.gameCardComponent, battle, player, freeCount)
        end
    end
end

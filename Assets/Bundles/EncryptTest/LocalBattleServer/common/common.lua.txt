function _G.IsSameWeekByTZAndOffset(time1, time2, tz, offsetDay)
    local offset = (7 - offsetDay) * 86400
    return GetWeekStartByTZ(time1 + offset, tz) == GetWeekStartByTZ(time2 + offset, tz)
end

function _G.GetDayStartByTZ(time, tz)
    return math_floor((time + tz * 3600) / 86400) * 86400 - tz * 3600
end

function _G.GetWeekStartByTZ(time, tz)
    local tz_tmp = 259200 + tz * 3600
    return math_floor((time + tz_tmp) / 604800) * 604800 - tz_tmp
end

--获取当天时间     start datetime (eg:15:25:36)
function _G.GetTodayStart( timestring)
    local time_offset = 0;
    if  timestring~=nil and string_len(timestring) > 0 then
        local patern = "(%d+):(%d+):(%d+)";
        local hour, min, sec = string_match(timestring, patern);
        if  hour then
            time_offset = time_offset + tonumber( hour)*3600;
        end
        if  min then
            time_offset = time_offset + tonumber( min)*60;
        end
        if  sec then
            time_offset = time_offset + tonumber( sec);
        end
    end
    return GetDayStart(GetCurTime()) + time_offset;
end

--获取当天时间         end datetime (eg:15:25:36)
function _G.GetTodayEnd( timestring)
    local time_offset = 86400;
    if  timestring~=nil and string_len(timestring) > 0 then
        time_offset = 0;
        local patern = "(%d+):(%d+):(%d+)";
        local hour, min, sec = string_match(timestring, patern);
        if  hour then
            time_offset = time_offset + tonumber( hour)*3600;
        end
        if  min then
            time_offset = time_offset + tonumber( min)*60;
        end
        if  sec then
            time_offset = time_offset + tonumber( sec);
        end
    end
    return GetDayStart(GetCurTime()) + time_offset;
end

function _G.SplitString(str, d) --str是需要查分的对象 d是分界符
    local lst = { }
    local len = string_len(str)--长度
    local index = 1
    while index <= len do
        local i = string_find(str, d, index) -- find 'next' 0
        if i == nil then
            table_insert(lst, string_sub(str, index, len))
            break
        end
        table_insert(lst, string_sub(str, index, i-1))
        if i == len then
            table_insert(lst, "")
            break
        end
        index = i + 1
    end
    return lst
end


-- 按同样权重随机1个元素
function _G.RandomOne(nodes)
    local weights = 0;
    local group = {};

    for _, node in pairs(nodes) do
        if node then
            group[#group + 1] = node;
            weights = weights + 1;
        end
    end
    if weights <= 0 then
        return;
    end
    local weight = math_random(weights);
    return group[weight];
end

-- 随机n个元素
function _G.RandomN(nodes, n)
    local group = {};
    for _, node in pairs(nodes) do
        if node then
            group[#group + 1] = node;
        end
    end

    if  n <= 0 or #group <=0 then return {}; end;
    if  n >= #group then return group end;

    local result = {};
    for i=1, n do
        local weight = math_random(#group);
        table.insert(result, group[weight]);
        table.remove(group, weight);
    end
    return result;
end

-- 按权重随机1个元素
function _G.RandomSize(sizes, sizeWeights)
    local weights = 0;
    local group = {};

    for id, weight in pairs(sizeWeights) do
        group[#group + 1] = {weight=weight, size=sizes[id]};
        weights = weights + weight;
    end

    if weights <= 0 then
        return
    end

    local weight = math_random(weights);

    for k, one in ipairs(group) do
        if weight <= one.weight then
            return one.size, k;
        else
            weight = weight - one.weight;
        end
    end
    return nil;
end

-- 按权重随机n个元素
function _G.RandomSizeN(sizes, sizeWeights, n)
    if  n <= 0 or #sizes <=0 then return end;
    if  n >= #sizes then return sizes end;

    local sizesLocal={};
    for k, v in pairs(sizes) do
        sizesLocal[#sizesLocal + 1] = v;
    end

    local weightsLocal ={};
    for k, v in pairs(sizeWeights) do
        weightsLocal[#weightsLocal + 1] = v;
    end

    local result = {};
    for i=1, n do
        local  v,k = RandomSize(sizesLocal, weightsLocal);
        result[#result+1] = v;
        table_remove(sizesLocal, k);
        table_remove(weightsLocal, k);
    end

    sizesLocal = nil;
    weightsLocal = nil;

    return result;
end

-- 按权重随机n个元素且返回元素的下标数组
function _G.RandomSizeNIndex(sizes, sizeWeights, n)
    local sizesCount = math_min(#sizes, #sizeWeights);
    if n <= 0 or sizesCount <= 0 then
        return {}, {};
    end

    local result = {};
    local resultIdx = {};
    if n >= sizesCount then
        for i = 1, sizesCount do
            result[i] = sizes[i]
            resultIdx[i] = i
        end
        return result, resultIdx
    end

    local sizesLocal = {};
    local totalWeight = 0;
    for i = 1, sizesCount do
        sizesLocal[i] = true;
        totalWeight = totalWeight + sizeWeights[i];
    end

    for k = 1, n do
        local weight = math_random(totalWeight);

        for i = 1, sizesCount do
            if sizesLocal[i] then
                if weight <= sizeWeights[i] then
                    result[#result + 1] = sizes[i];
                    resultIdx[#resultIdx + 1] = i;
                    sizesLocal[i] = false;
                    totalWeight = totalWeight - sizeWeights[i];
                    break;
                else
                    weight = weight - sizeWeights[i];
                end
            end
        end
    end

    return result, resultIdx;
end

-- 洗牌，等概率
function _G.Shuffle(array)
    if type(array) ~= "table" then
        return array;
    end
    local size = #array;
    local rand;
    for i = size, 1, -1 do
        rand = math_random(size);
        array[i], array[rand] = array[rand], array[i];
    end
    return array;
end

-- 洗牌，带权重，数组node自带属性weight
function _G.ShuffleWeight(arrayWithWeight)
    if type(arrayWithWeight) ~= "table" then
        return arrayWithWeight;
    end

    local sumWeight = 0;
    for i, node in ipairs(arrayWithWeight) do
        if not node.weight then
            return arrayWithWeight;
        end
        sumWeight = sumWeight + node.weight;
    end

    local randWeight;
    local node;
    local size = #arrayWithWeight;

    for i = 1, size do
        randWeight = math_random(sumWeight);
        for j = i + 1, size do
            node = arrayWithWeight[j];
            if randWeight <= node.weight then
                arrayWithWeight[i], arrayWithWeight[j] = node, arrayWithWeight[i];
                break;
            else
                randWeight = randWeight - node.weight;
            end
        end
        sumWeight = sumWeight - arrayWithWeight[i].weight;
    end

    return arrayWithWeight;
end

-- 洗牌，带权重，数组和权重数组分开
function _G.Shuffleweight2(array, arrayWeight)
    if type(array) ~= "table" or type(arrayWeight) ~= "table" or #array > #arrayWeight then
        return array;
    end

    local sumWeight = 0;
    for i, node in ipairs(array) do
        sumWeight = sumWeight + arrayWeight[i];
    end

    local randWeight;
    local nodeWeight;
    local size = #array;

    for i = 1, size do
        randWeight = math_random(sumWeight);
        for j = i + 1, size do
            nodeWeight = arrayWeight[j];
            if randWeight <= nodeWeight then
                array[i], array[j] = array[j], array[i];
                arrayWeight[i], arrayWeight[j] = nodeWeight, arrayWeight[i];
                break;
            else
                randWeight = randWeight - nodeWeight;
            end
        end
        sumWeight = sumWeight - arrayWeight[i];
    end

    return array, arrayWeight;
end

-- 深拷贝，保留引用和元表
function _G.DeepClone( obj )
    local inTable = {};
    local function func(obj)
        if type(obj) ~= "table" then  --判断表中是否有表
            return obj;
        elseif inTable[obj] then  --已复制，返回引用，使原表相同的引用复制后仍相同
            return inTable[obj];
        end
        local newTable = {};  --定义一个新表
        inTable[obj] = newTable;
        for k,v in pairs(obj) do  --复制旧表的key和Value
            newTable[func(k)] = func(v);
        end
        return setmetatable(newTable, getmetatable(obj))  --保留元表
    end
    return func(obj) --递归复制表中的表
end
-- 别名
_G.DeepCopy = _G.DeepClone;

function table.tablen(tab)
    local len = 0
    for k,v in pairs(tab) do
        len = len + 1
    end

    end

    local ret_str = nil
    for idx, key in pairs(t) do
        if not ret_str then
            ret_str = tostring(key)
        else
            ret_str = string_format("%s%s%s", ret_str, c, key)
        end
    end

    return ret_str
end

    end

-- 分数计算
function _G.ScoreAndRankCal(serverType, battleScore, battleResult, winningStreak, calScore)
    local nowRankSeasonConf = _G.GetSeasonConf(serverType);

    if not calScore then
        return 0, 0;
    end

    local rankConf = _G.GetRank(serverType, battleScore);
    if not rankConf then
        return 0, 0;
    end

    local rankRule = _G.PvpRankRuleTable[nowRankSeasonConf.seasonRankRuleId];

    local updateScore = 0;
    if battleResult == BATTLE_END_RESULT_WIN then
        -- 胜利的情况
        winningStreak  = winningStreak + 1;
        local winScore = rankConf.winScore;
        updateScore    = winScore + math.min(rankRule.winingMaxScore, rankRule.winingScore * (winningStreak - 1));
    elseif battleResult == BATTLE_END_RESULT_LOST then
        -- 失败的情况
        winningStreak   = 0;
        local lostScore = rankConf.lostScore
        updateScore     = -lostScore;
    else
        -- 异常情况, 数据不做修改
    end
    return updateScore, winningStreak;
end

-- 身份证号计算年龄
function _G.AgeCal(idCard)
    if idCard and #idCard >= 14 then
        local birthdayStr = string.sub(idCard, 7, 14)
        local year = tonumber(string.sub(birthdayStr, 1, 4))
        local month = tonumber(string.sub(birthdayStr, 5, 6))
        local day = tonumber(string.sub(birthdayStr, 7, 8))

        local today = os.date("*t")
        local age = today.year - year

        if today.month < month or (today.month == month and today.day < day) then
            age = age - 1
        end

        return age
    else
        LogDebug("id card is invalid, idCard:{}", idCard)
        return -1
    end
end

local versionSub = {}
local versionBit = {3, 3, 4, 6}
local versionNum = #versionBit
-- @param versionStr 版本字符串可以是包含数字的
-- @param ceil 值为缺省时自动填入可能的最大值 例如 versionStr='0.10', ceil=true => 000 00a fff fffffff
function _G.versionAtoi(versionStr, ceil)
    local index = 0
    local versioInt = 0
    for i=1, versionNum do
        if not ceil then
            versionSub[i] = 0
        else
            versionSub[i] = (1 << versionBit[i]) - 1
        end
    end
    for v in string_gmatch(versionStr, "(%d+)") do
        index = index + 1
        versionSub[index] = math_tointeger(v)
    end
    for i=1, versionNum do
        versioInt = versioInt * (10 ^ versionBit[i])
        versioInt = versioInt + versionSub[i]
    end
    return versioInt
end

    end
    return len
end

--将客户端版本号转换成int,方便判断大小.
function GetIntClientVer(ver)
    if type(ver) ~= "string" then
        return 0
    end

    local res = 0
    local para = {10000000,10000,1}

    local idx = 1
    for v in string_gmatch(ver, "%d+") do
        if not para[idx] then
            break
        end

        res = res + v * para[idx]
        idx = idx + 1
    end

    return res
end

--是否打印协议参数
local printProtoOutputPara = false
--打印table参数最大层数,设置为0则关闭打印table
local PrintProtoMaxPrintParaRecursive = 2
--打印协议的过滤列表
local PrintProtoFiltrationList = {
    ["HeartBeatRequest"] = true,
    ["HeartBeatResponse"] = true,
    ["..."] = true,
}

function PrintTab(tab, idx)
    if type(tab) ~= "table" then
        return "nil"
    end

    if idx >= PrintProtoMaxPrintParaRecursive then
        return "..."
    end

    local ty
    local str = ""
    for k,v in pairs(tab) do
        ty = type(v)
        if ty == "nil" then
            str = string_format("%s [%s]=nil ", str, k)
        elseif ty == "boolean" then
            str = string_format("%s [%s]=%s ", str, k, (v and "true" or "false"))
        elseif ty == "number" then
            str = string_format("%s [%s]=%s ", str, k, v)
        elseif ty == "string" then
            str = string_format("%s [%s]=%s ", str, k, v)
        elseif ty == "table" then
            str = string_format("%s [%s]={%s} ", str, k, PrintTab(v, idx+1))
        else
            str = string_format("%s [%s]=%s ", str, ty, ty)
        end
    end
    return str
end

function PrintVarargs(...)
    local ty
    local str = ""
    for k,v in ipairs({...}) do
        ty = type(v)
        if ty == "nil" then
            str = string_format("%s (nil) ", str)
        elseif ty == "boolean" then
            str = string_format("%s (bol)=%s ", str, (v and "true" or "false"))
        elseif ty == "number" then
            str = string_format("%s (num)=%d ", str, v)
        elseif ty == "string" then
            str = string_format("%s (str)=%s ", str, v)
        elseif ty == "table" then
            str = string_format("%s (tab)={%s} ", str, PrintTab(v, 1))
        else
            str = string_format("%s (%s) ", str, ty)
        end
    end

    return str
end

    if PrintProtoFiltrationList[msg] then

    end

    if not printProtoOutputPara then
        LogInfo("[proto] ss[{}] {} msg[{}]", uid, ty, msg)
    else
        LogInfo("[proto] ss[{}] {} msg[{}] para[{}]", uid, ty, msg, PrintVarargs(...))
    end
end

    return string.gsub(template, '[xy]', function (c)

---@param battleTeams MatchBattleTeam[]
function RebuildCoopBattleTeams(battleTeams)
    local newBattleTeam = {
        players = {},
        rating = 0,
        count = 0,
    }
    local totalRating = 0
    local totalCount = 0
    for idx, battleTeam in pairs(battleTeams) do
        totalCount = totalCount + battleTeam.count
        for _, player in pairs(battleTeam.players) do
            newBattleTeam.players[player.id] = player
        end
    end

    newBattleTeam.count = totalCount
    newBattleTeam.rating = totalRating / totalCount

    return { newBattleTeam }
end

-- 准备战斗
function NoticeBattle(mode, version, battleTeams)
    -- count  队伍的数量
    -- score 队伍分
    -- players
    -- score: score
    -- id: uid
    -- svr: 玩家所在lobby
    -- 通知战斗服开始初始化流程
    -- 随机选择战斗服

    -- 根据战斗mode重构battleTeams
    if mode == _G.BattleGameTypeMode.BATTLE_GAME_TYPE_MODE_COOP then
        battleTeams = RebuildCoopBattleTeams(battleTeams)
    end

    -- 统一处理一下playerIndex
    for _, team in pairs(battleTeams) do
        local idx = 1
        for _, player in pairs(team.players) do
            player.playerIndex = idx;
            idx = idx + 1;
        end
    end

    local instance = GetLeastBattleServer()
    if instance then
        message.SendP2PMessage("battle", instance, "PrepareBattle", mode, version, battleTeams);

        _G.BattleId2BattleCount[instance] = (_G.BattleId2BattleCount[instance] or 0) + 1
        return instance;
    end
    LogErr("no battle server exist!!!")
end

-- 获取到战斗场次最少的战斗服
function GetLeastBattleServer()
    local instances = _G.diveSDMgr.get_all_instance_id("battle");
    if not instances then
        return
    end

    local leastBattleServer
    local leastBattleCount
    for _, instance in pairs(instances) do
        local battleCount = _G.BattleId2BattleCount[instance] or 0
        if not leastBattleCount or leastBattleCount > battleCount then
            leastBattleServer = instance
            leastBattleCount  = battleCount
        end
    end

    LogDebug("GetLeastBattleServer _G.BattleId2BattleCount={} leastBattleServer={}", _G.BattleId2BattleCount, leastBattleServer)

    return leastBattleServer
end

-- 用于问道玩法计算法宝属性
function CalWenDaoCardAttribute(sutraCard)
    local sutraConfig = _G.SutraTable[sutraCard.id]
    if not sutraConfig then
        LogErr("CalWenDaoCardAttribute, invalid card, id:{}", sutraCard.id)
        return nil
    end

    -- 基础等级属性
    local sutraLevelTypeGroup = _G.SutraLevelTable[sutraConfig.levelType]
    if not sutraLevelTypeGroup then
        LogErr("CalWenDaoCardAttribute, invalid card, id:{} levelType:{}", sutraCard.id, sutraConfig.levelType)
        return nil
    end
    local sutraLevelConfig = sutraLevelTypeGroup[sutraCard.level]
    if not sutraLevelConfig then
        LogErr("CalWenDaoCardAttribute, invalid card, id:{} levelType:{} level:{}", sutraCard.id, sutraConfig.levelType, sutraCard.level)
        return nil
    end

    -- 品质倍率
    local sutraQualityTypeConfig = _G.SutraQualityTypeTable[sutraConfig.qualityType]
    if not sutraQualityTypeConfig then
        LogErr("CalWenDaoCardAttribute, can not find sutra quality type config, quality:{}", sutraCard.quality)
        return nil
    end
    local qualityRate = sutraQualityTypeConfig.qualityAttr[sutraCard.quality] or sutraQualityTypeConfig.qualityAttr[#sutraQualityTypeConfig.qualityAttr]

    -- levelParam
    local levelParamConfig = _G.SutraLevelParamTable[sutraConfig.levelParamType]
    if not levelParamConfig then
        LogErr("CalWenDaoCardAttribute, can not find sutra level param config, levelParam:{}", sutraConfig.levelParamType)
        return nil
    end

    -- 计算基础属性
    local attributes = {}
    attributes.baseAttack = sutraLevelConfig.baseAttack * qualityRate / 10000 * levelParamConfig.baseAttack / 10000
    attributes.baseAttackSpeed = sutraLevelConfig.baseAttackSpeed * levelParamConfig.baseAttackSpeed / 10000
    attributes.baseCritRate = sutraLevelConfig.baseCritRate * levelParamConfig.baseCritRate / 10000
    attributes.baseCritDamage = sutraLevelConfig.baseCritDamage * levelParamConfig.baseCritDamage / 10000
    attributes.baseTowerHp = sutraLevelConfig.baseTowerHp * qualityRate / 10000 * levelParamConfig.baseTowerHp / 10000

    -- 如果没有数值的属性 则从属性表中获取基础数值
    for _, data in pairs(_G.UnitAttributeTable) do
        attributes[data.codename] = attributes[data.codename] or data.baseValue
    end

    return attributes
end

-- 用于问道玩法计算元神属性
function CalWenDaoSpiritAttribute(spiritData, mapAttr)
    local attrs = {}
    local sutraHp = 0
    local sutraAtk = 0
    for _, sutraCard in pairs(spiritData.sutras or {}) do
        local attributes = CalWenDaoCardAttribute(sutraCard)
        sutraHp = sutraHp + attributes.baseTowerHp
        sutraAtk = sutraAtk + attributes.baseAttack
    end

    attrs[_G.SpiritAttributeType.SPIRIT_ATTR_HP] = sutraHp
    attrs[_G.SpiritAttributeType.SPIRIT_ATTR_ATK] = sutraAtk

    -- 基础属性
    local spiritConfig = _G.SpiritTable[spiritData.spiritId]
    if not spiritConfig then
        LogErr("CalSpiritAttribute spiritConfig is nil, spiritId={}", spiritData.spiritId)
    else
        for attrId, attrValue in pairs(spiritConfig.defaultAffix) do
            attrs[attrId] = (attrs[attrId] or 0) + attrValue
        end
    end

    local spiritStarConfig = _G.SpiritStarTable[spiritData.spiritId][spiritData.level]
    if not spiritStarConfig then
        LogErr("CalSpiritAttribute spiritConfig is nil, spiritId={}, level={}", spiritData.spiritId, spiritData.level)
    else
        for attrId, attrValue in pairs(spiritStarConfig.spiritSelfAttrAdds) do
            attrs[attrId] = (attrs[attrId] or 0) + attrValue
        end
    end

    -- 加点属性
    for id, point in pairs(spiritData.affixPoints) do
        local affixConfig = _G.SpiritAffixTable[id]
        if not affixConfig then
            LogErr("CalSpiritAttribute affixConfig is nil, id={}", id)
        else
            attrs[affixConfig.attr] = (attrs[affixConfig.attr] or 0) + affixConfig.levelValue * point
        end
    end

    --地图加成
    if mapAttr and mapAttr[spiritData.spiritId] then
        for _, attrData in pairs(mapAttr[spiritData.spiritId]) do
            attrs[attrData.attr] = (attrs[attrData.attr] or 0) + attrData.value;
        end
    end

    -- level2属性
    local spiritLevelGroup = _G.SpiritLevelTable[spiritConfig.spiritLevelGroup] or {}
    local spiritLevelConfig = spiritLevelGroup[spiritData.level3 or 1]
    if not spiritLevelConfig then
        LogErr("CalSpiritAttribute spiritLevelConfig is nil, spiritId={}, group={} level3={}", spiritData.spiritId, spiritConfig.spiritLevelGroup, spiritData.level3)
    else
        attrs[_G.SpiritAttributeType.SPIRIT_ATTR_HP_FIXED] = (attrs[_G.SpiritAttributeType.SPIRIT_ATTR_HP_FIXED] or 0) + spiritLevelConfig.hp * spiritConfig.levelHpRate / 10000
        attrs[_G.SpiritAttributeType.SPIRIT_ATTR_ATK_FIXED] = (attrs[_G.SpiritAttributeType.SPIRIT_ATTR_ATK_FIXED] or 0) + spiritLevelConfig.atk * spiritConfig.levelAtkRate / 10000
    end

    return attrs
end

-- 计算战力
-- 元神战斗力需要计算元神身上大部分的属性
-- 攻击战力1：	(最终攻击力*攻击力系数)*(速度/速度系数)
-- 攻击战力2：	攻击战力1*(1+连击几率)
-- 攻击战力3：	攻击战力2*(1+命中率)
-- 攻击战力4：	攻击战力3*[最终暴击率*最终暴击伤害+1]
-- 攻击战力5：	攻击战力4*(1+吸血抗性百分比)*(1+击晕抗性百)
-- 最终攻击战力：	攻击战力5*(1+反击几率)
-- 道心战力1：	最终道心*道心系数
-- 道心战力2：	道心战力1*（1+连击抗性）
-- 道心战力3：	道心战力2*（1+闪避率）
-- 道心战力4：	道心战力3*（1+暴击抗性*基础暴击伤害）
-- 最终道心战力：	道心战力4*(1+吸血百分比)*(1+击晕几率)
-- 元神最终战力：	最终攻击战力+最终道心战力
function CalWenDaoBattleScore(wenDaoTeam)
    local totalBattleScore = 0

    local getAttr = function(attrs, attrId)
        return attrs[attrId] or 0
    end

    for _, spiritData in pairs(wenDaoTeam or {}) do
        local battleScore = 0
        local attrs = CalWenDaoSpiritAttribute(spiritData)
        local attackFinal = getAttr(attrs, _G.SpiritAttributeType.SPIRIT_ATTR_ATK)
            * (1 + getAttr(attrs, _G.SpiritAttributeType.SPIRIT_ATTR_ATK_RATE) / 10000)
        local attackCombat1 = (attackFinal * _G.CommonValue.wdAtkRate) *
            (getAttr(attrs, _G.SpiritAttributeType.SPIRIT_ATTR_SPEED_FINAL) / _G.CommonValue.wdSpeedRate)
        local attackCombat2 = attackCombat1 * (1 + getAttr(attrs, _G.SpiritAttributeType.SPIRIT_ATTR_COMBO_FINAL) / 10000)
        local attackCombat3 = attackCombat2 * (1 + getAttr(attrs, _G.SpiritAttributeType.SPIRIT_ATTR_HIT_RATE_FINAL) / 10000)
        local attackCombat4 = attackCombat3 * (1 + getAttr(attrs, _G.SpiritAttributeType.SPIRIT_ATTR_CRIT_RATE_FINAL) / 10000 * getAttr(attrs, _G.SpiritAttributeType.SPIRIT_ATTR_CRIT_DAMAGE_FINAL) / 10000)
        local attackCombat5 = attackCombat4 * (1 + getAttr(attrs, _G.SpiritAttributeType.SPIRIT_ATTR_ANTI_LIFESTEAL_RATE_FINAL) / 10000) * (1 + getAttr(attrs, _G.SpiritAttributeType.SPIRIT_ATTR_ANTI_STUN_RATE_FINAL) / 10000)
        local attackCombatFinal = attackCombat5 * (1 + getAttr(attrs, _G.SpiritAttributeType.SPIRIT_ATTR_COUNTER_RATE_FINAL) / 10000)

        local hpFinal = getAttr(attrs, _G.SpiritAttributeType.SPIRIT_ATTR_HP)
            * (1 + getAttr(attrs, _G.SpiritAttributeType.SPIRIT_ATTR_HP_RATE) / 10000)
        local hpCombat1 = hpFinal * _G.CommonValue.wdHpRate
        local hpCombat2 = hpCombat1 * (1 + getAttr(attrs, _G.SpiritAttributeType.SPIRIT_ATTR_ANTI_COMBO_FINAL) / 10000)
        local hpCombat3 = hpCombat2 * (1 + getAttr(attrs, _G.SpiritAttributeType.SPIRIT_ATTR_DODGE_RATE_FINAL) / 10000)
        local hpCombat4 = hpCombat3 * (1 + getAttr(attrs, _G.SpiritAttributeType.SPIRIT_ATTR_ANTI_CRIT_RATE_FINAL) / 10000 * getAttr(attrs, _G.SpiritAttributeType.SPIRIT_ATTR_CRIT_DAMAGE) / 10000)
        local hpCombatFinal = hpCombat4 * (1 + getAttr(attrs, _G.SpiritAttributeType.SPIRIT_ATTR_LIFESTEAL_RATE_FINAL) / 10000) * (1 + getAttr(attrs, _G.SpiritAttributeType.SPIRIT_ATTR_STUN_RATE_FINAL) / 10000)
        battleScore = attackCombatFinal + hpCombatFinal

        totalBattleScore = totalBattleScore + battleScore
    end

    return totalBattleScore
end

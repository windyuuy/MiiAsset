-- 阵营击杀怪物

local buffComponent = import("battle/components/buffComponent.lua")
local eventManager = import("battle/event/eventManager.lua")

function buffTrigger(component, battle, buff, triggerTypeIndex)
    local callback = function(listener, properties, event)
        local teamId = event.attacker.teamId
        local buffOwner = properties.buff.owner
        local campType = properties.campType
        if campType == 1 then
            -- 己方击杀
            if buffOwner.teamId ~= teamId then
                return false
            end
        elseif campType == 2 then
            -- 敌方击杀
            if buffOwner.teamId == teamId then
                return false
            end
        end

        listener.count = (listener.count or 0) + 1
        if listener.count < properties.perCount then
            return false
        end

        -- 重置计数器
        listener.count = 0
        buffComponent.buffEffect(properties.buffComponent, properties.buff)

        return true
    end

    local buffConfig = _G.BuffTable[buff.buffId]
    local triggerTypeData = buffConfig.triggerTypeDatas[triggerTypeIndex]
    local campType = triggerTypeData[1]
    local perCount = triggerTypeData[2]

    local properties = {
        buff            = buff,
        attackerId      = buff.owner.entryId,
        buffComponent   = component,
        campType        = campType,
        perCount        = perCount
    }
    local listenerId = eventManager.AddEventListener(battle.eventManager,
        _G.EventType.KILL_MONSTER, callback, properties, false)
    buff.listenerEventType = _G.EventType.KILL_MONSTER
    table.insert(buff.listenerIds, listenerId)
end

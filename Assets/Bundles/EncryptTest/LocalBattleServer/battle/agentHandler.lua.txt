_G.AgentTable = _G.AgentTable or {}

_G.GetAgentByUid = function(uid)
    return _G.AgentTable[uid];
end
_G.RemoveAgnetByUid = function(uid)
    _G.AgentTable[uid] = nil;
end

-- agent收到连接请求
function s2s.JoinBattleRequest(svr, token, param)
    local battleToken = param.token;
    if not battleToken then
        message.SendP2PMessage("battleAgent", svr, "BattleMessage", uid, token, "BattleLogic_JoinBattleResponse", {errCode=errCode.BATTLE_ERROR_NO_BATTLE_ID});
        return;
    end

    -- token反向关联uid
    local uid = battleMgr.GetPlayerTokenUid(battleToken);

    -- 获取该token的battle_id
    local battleId = battleMgr.GetUidTokenBattleId(uid, battleToken);
    if not battleId then
        message.SendP2PMessage("battleAgent", svr, "BattleMessage", uid, token, "BattleLogic_JoinBattleResponse", {errCode=errCode.BATTLE_ERROR_NO_BATTLE_ID});
        return;
    end

    local agent = {
        uid = uid,
        battleId    = battleId,
        battleToken = battleToken,
        agentSvr    = svr,
        token       = token,
        Call = tools.LogDecorator(function(msg, data)
            message.SendP2PMessage("battleAgent", svr, "BattleMessage", uid, token, msg, data);
        end);
    }
    _G.AgentTable[uid] = agent

    local res = battleMgr.JoinBattle(agent, battleId);
    agent.Call("BattleLogic_JoinBattleResponse", {errCode=res, battleId=battleId});
end

-- battleAgent转发过来的消息
function s2s.BattleAgentMessage(svr, uid, battleId, msg, param)
    -- LogInfo("BattleAgentMessage msg={} param={}", msg, param)
    local agent = _G.AgentTable[uid]
    if not agent or agent.battleId ~= battleId then
        return
    end

    -- 转发到c2s消息里面
    local proc = _G.c2s[msg];
    if not proc then
        LogErr("undefined msg {} from session {}", msg, token);
        return;
    end

    local ok, err = xpcall(proc, debug.traceback, agent, param);
    if not ok then
        LogErr("failed to call msg c2s.{}", msg);
        LogErr("{}", err);
    end

    agent.lastRequest = {
        msg = msg,
        param = param,
    }
end

-- 战斗投降
function c2s.BattleLogic_GiveupBattleRequest(ss, param)
    local activeBattleId = battleMgr.GetUidActiveBattleId(ss);
    if not activeBattleId then
        ss.Call("BattleLogic_GiveupBattleResponse", {errCode=errCode.BATTLE_ERROR_NO_BATTLE_ID});
        return;
    end

    local battle = battleMgr.GetBattle(activeBattleId)
    if not battle then
        ss.Call("BattleLogic_GiveupBattleResponse", {errCode=errCode.BATTLE_ERROR_NO_BATTLE_ID})
        return
    end

    if battle.status ~= _G.BattleStatus.PLAY then
        ss.Call("BattleLogic_GiveupBattleResponse", {errCode=errCode.BATTLE_ERROR_NO_FIGHT_STATUS})
        return
    end

    LogDebug("c2s.BattleLogic_GiveupBattleRequest player:{} give up", ss.uid)

    -- 放弃战斗
    local resp = battleMgr.GiveUpBattle(ss.uid, activeBattleId);
    ss.Call("BattleLogic_GiveupBattleResponse", resp);
end

-- 客户端已连接战斗服，并且准备好开始战斗
function c2s.BattleLogic_BattleReadyRequest(ss, param)
    --LogInfo("BattleLogic_BattleReadyRequest ss:{} param: {}", ss, param);
    local battle = battleMgr.GetBattle(param.battleId)
    if not battle then
        ss.Call("BattleLogic_BattleReadyResponse", {errCode=errCode.BATTLE_ERROR_NO_BATTLE_ID, battleId=param.battleId});
        return;
    end

    -- 先根据状态判断是开始还是重连
    if battle.status < _G.BattleStatus.PLAY then
        ss.battleId = param.battleId;
        local res = battleMgr.ReadyBattle(ss, param.battleId);
        ss.Call("BattleLogic_BattleReadyResponse", {errCode=res, battleId=param.battleId});
    else
        -- TODO: 重连数据
    end
end

function s2s.ReconnectBattle(svr, uid, token, param)
    local battleToken = param.token;
    if not battleToken then
        message.SendP2PMessage("battleAgent", svr, "BattleMessage", uid, token, "BattleLogic_ReconnectBattleResponse", {errCode=errCode.BATTLE_ERROR_NO_BATTLE_ID});
        return;
    end

    -- 检查uid和token
    local battleId = battleMgr.GetUidTokenBattleId(uid, battleToken);
    if not battleId then
        message.SendP2PMessage("battleAgent", svr, "BattleMessage", uid, token, "BattleLogic_ReconnectBattleResponse", {errCode=errCode.BATTLE_ERROR_NO_BATTLE_ID});
        return;
    end

    local player = battleMgr.GetPlayer(battleId, uid)

    local agent = {
        uid = uid,
        battleId    = battleId,
        battleToken = battleToken,
        agentSvr    = svr,
        Call = tools.LogDecorator(function(msg, data)
            message.SendP2PMessage("battleAgent", svr, "BattleMessage", uid, token, msg, data);
        end)
    }
    _G.AgentTable[uid] = agent
    local battle = battleMgr.GetBattle(battleId);
    local intent = battleIntent.GetBattleIntent(battleId)

    agent.Call("BattleLogic_ReconnectBattleResponse", {
        errCode = errCode.SUCCESS,
        serverTime = _G.GetCurTimeMS(),
        playerStatus = player.status,
        battleData = battle:GetBattleData(),
        frameId = intent.frameId,
    });

    battle:NotifyPreBattleData(uid)
end

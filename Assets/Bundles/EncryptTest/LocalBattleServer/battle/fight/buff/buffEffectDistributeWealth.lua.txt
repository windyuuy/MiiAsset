-- 18. 均分财富
local battleMgr = import("battle/battleMgr.lua")
local resourceComp = import("battle/components/resourceComponent.lua")

-- buff生效
---@param buff Buff
function BuffEffect(buff)
    -- 目前只支持1v1对战！！！！！！

    local buffConfig = _G.BuffTable[buff.buffId]

    ---@type Battle
    local battle = battleMgr.GetBattle(buff.owner.battleId)
    local releaseRate = buffConfig.effectData[1]
    local targetRate = buffConfig.effectData[2]

    local total = 0
    for _, player in pairs(battle.players) do
        total = total + player.resourceComponent.mana
    end

    local releaseCount = total * releaseRate / (releaseRate + targetRate)
    local targetCount = total - releaseCount

    for _, player in pairs(battle.players) do
        if player.teamId == buff.owner.teamId then
            resourceComp.ChangeMana(player.resourceComponent, player, targetCount - player.resourceComponent.mana)
        else
            resourceComp.ChangeMana(player.resourceComponent, player, releaseCount - player.resourceComponent.mana)
        end
    end
end

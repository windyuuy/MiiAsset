local message = import("common/message.lua");
local robotUtil = import("common/robotUtil.lua");

_G.CallOtherUser = function(uid, toUid, offMsgType, msg, ...)
    if robotUtil.IsRobotUid(toUid) then
        return true;
    end

    if _G.uidTable and _G.uidTable[toUid] then
        local selfSvrId = _G.diveSDMgr.get_local_instance_id();

        local func = _G.s2s[msg];
        func(selfSvrId, toUid, uid, selfSvrId, offMsgType, ...);

        return true;
    end

    local instatus = _G.diveSDMgr.get_random_instance_id("online");
    if instatus then
        -- 请求online服务，申请切磋
        message.SendP2PMessage("online", instatus, "RouterToUser", uid, toUid, offMsgType, msg, ...);
        return true;
    end

    return false;
end

_G.CallLobbyMessage = function(req, ...)
    -- 获取lobby服
    local instance = _G.diveSDMgr.get_random_instance_id("lobby");
    if not instance then
        LogErr("CallLobbyMessage no lobby server!!!")
        return false;
    end

    -- 请求lobby服
    return message.SendP2PMessage("lobby", instance, req, ...);
end

_G.CallAllLobbyMessage = function(req, ...)
    -- 获取lobby服
    local instances = _G.diveSDMgr.get_all_instance_id("lobby");
    if not instances then
        LogErr("CallAllLobbyMessage no lobby server!!!")
        return false;
    end

    for _, instance in pairs(instances) do
        -- 请求lobby服
        message.SendP2PMessage("lobby", instance, req, ...);
    end
    return true;
end

    local instances = _G.diveSDMgr.get_all_instance_id("rank");
    if not instances then
        LogErr("CallRankMessage no rank server!!!")
        return false;
    end

    local local_instance_id = _G.diveSDMgr.get_local_instance_id();
    for _, instance in pairs(instances) do
        -- 请求rank服
        if instance ~= local_instance_id then
            message.SendP2PMessage("rank", instance, req, ...);
        end
    end
    return true;
end

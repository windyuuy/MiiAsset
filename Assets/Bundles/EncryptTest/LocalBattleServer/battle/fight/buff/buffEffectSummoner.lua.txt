-- 20. 召唤召唤物
local battleMgr = import("battle/battleMgr.lua")

-- buff生效
---@param buff Buff
function BuffEffect(buff)
    local battle      = battleMgr.GetBattle(buff.owner.battleId)
    local buffConfig  = _G.BuffTable[buff.buffId]
    local owner       = buff.owner
    local summonCount = buffConfig.effectData[1]    -- 召唤数量
    local summonerId  = buffConfig.effectData[2]    -- 召唤物id
    --LogDebug("buffEffectSummoner:BuffEffect: buffId={} summonCount={} summonerId={} entryId={}", buff.buffId, summonCount, summonerId, buff.owner.entryId)

    buff.summoners = buff.summoners or {}

    summonCount = owner.addSummonerCount and summonCount + owner.addSummonerCount or summonCount

    local siteId = 0
    if owner:GetEntryType() == _G.EntryType.SUTRA then
        local player = battle.players[owner.playerId]
        siteId = player.playerIndex
    elseif owner:GetEntryType() == _G.EntryType.TOWER then
        siteId = 1
    else
        LogErr("buffEffectSummoner:BuffEffect: owner type error, entryId={} entryType={}", owner.entryId, owner:GetEntryType())
        return
    end

    for _ = 1, summonCount do
        local entryId = battle:CreateSummoner(buff.owner.teamId, summonerId, buff.owner, siteId)
        table.insert(buff.summoners, entryId)
    end
end

-- 取消buff
function UnBuffEffect(buff)
    local battle = battleMgr.GetBattle(buff.owner.battleId)
    for _, summoner in pairs(buff.summoners) do
        battle:DestroySummoner(buff.owner.teamId, summoner.entryId)
    end
end

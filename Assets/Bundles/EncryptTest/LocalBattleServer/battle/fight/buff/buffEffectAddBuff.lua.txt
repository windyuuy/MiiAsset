-- 8. buff添加子buff的效果
local buffComp = import("battle/components/buffComponent.lua")
local battleMgr = import("battle/battleMgr.lua")

-- buff生效
---@param buff Buff
function BuffEffect(buff)
    local battle     = battleMgr.GetBattle(buff.owner.battleId)
    local buffConfig = _G.BuffTable[buff.buffId]

    for i = 1, #buffConfig.effectData, 2 do
        local buffId = buffConfig.effectData[i]
        local layer  = buffConfig.effectData[i + 1]
        buffComp.AddBuff(battle, buffId, buff.owner, buff.origin, layer, buff.skillId)
    end
end

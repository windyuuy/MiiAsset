-- 31. 怪物被秒杀时触发

local buffComponent = import("battle/components/buffComponent.lua")
local eventManager = import("battle/event/eventManager.lua")

function buffTrigger(component, battle, buff, triggerTypeIndex)
    local callback = function(listener, properties, event)
        if event.teamId ~= properties.teamId then
            return false
        end

        buffComponent.buffEffect(properties.buffComponent, properties.buff, event.monsterId)
        return true
    end
    local properties = { teamId = buff.owner.teamId, buff = buff, buffComponent = component }
    local listenerId = eventManager.AddEventListener(battle.eventManager, _G.EventType.MONSTER_BE_EXECUTE, callback, properties, false)
    buff.listenerEventType = _G.EventType.MONSTER_BE_EXECUTE
    table.insert(buff.listenerIds, listenerId)
end

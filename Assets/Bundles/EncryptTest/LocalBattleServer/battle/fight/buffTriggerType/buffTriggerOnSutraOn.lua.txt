-- 场上指定法宝上阵

local buffComponent = import("battle/components/buffComponent.lua")
local eventManager = import("battle/event/eventManager.lua")

function buffTrigger(component, battle, buff, triggerTypeIndex)
    local callback = function(listener, properties, event)
        local teamId = event.teamId
        if properties.teamId ~= teamId then
            return false
        end

        if not table.contains(event.sutraIds, properties.targetSutraId) then
            return false
        end

        buffComponent.buffEffect(properties.buffComponent, properties.buff)
        return true
    end
    local buffConfig = _G.BuffTable[buff.buffId]
    local triggerTypeData = buffConfig.triggerTypeDatas[triggerTypeIndex]
    local targetSutraId = triggerTypeData[1]
    local properties = { teamId = buff.owner.teamId, buff = buff, buffComponent = component, targetSutraId = targetSutraId }
    local listenerId = eventManager.AddEventListener(battle.eventManager, _G.EventType.SUTRA_ON, callback, properties, false)
    buff.listenerEventType = _G.EventType.SUTRA_ON
    table.insert(buff.listenerIds, listenerId)
end

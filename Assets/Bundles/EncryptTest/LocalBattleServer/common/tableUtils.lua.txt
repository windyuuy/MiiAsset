        if not ok then

        end
    end

    --            local loadDependFunc = _G.TableDependFunctionMap[tableName];
    --            loadDependFunc();
    --        end
    --    end
    --end

    _G.TABLE_INITED = true;

    for key, callbackFunc in pairs(_G.TableCallback) do
        local ok, err = xpcall(callbackFunc, debug.traceback);
        if not ok then
            LogErr("TableCallback error: {}", err);
        end
    end

    _G.TableCallback = {};

    LogInfo("load table success");
end

function LoadTable(tableName)
    local loadFunc = _G.TableFunctionMap[tableName];
    if not loadFunc then
        return
    end

    local ok, err = xpcall(loadFunc, debug.traceback);
    if not ok then
        LogErr("load table error:{}, tableName:{}", err, tableName);
    else
        local loadAfterNames = _G.TableLoadAfter[tableName];
        for _, afterLuaName in pairs(loadAfterNames or {}) do
            dive.Reload("tables/" .. afterLuaName)
        end
    end
end

-- 赛季表
function LoadSeasonTimeTable()
    _G.SeasonFutureTable    = {};
    _G.LastSeason           = {};
    local lastSeasonEndTime = {};
    local seasonTimeTable   = {}
    local curr              = _G.GetCurTime();

    for id, data in pairs(_G.SeasonTimeTable) do
        data.startTime = _G.GetUnixTimeStamp(data.seasonStartData)
        if data.seasonStopData == "" then
            data.stopTime = 0
        else
            data.stopTime  = _G.GetUnixTimeStamp(data.seasonStopData)
        end

        seasonTimeTable[data.serverType] = seasonTimeTable[data.serverType] or {}
        seasonTimeTable[data.serverType][id] = data

        if data.stopTime > 0 and data.stopTime < curr then
            if data.stopTime > (lastSeasonEndTime[data.serverType] or -1) then
                lastSeasonEndTime[data.serverType] = data.stopTime;
                _G.LastSeason[data.serverType]     = id;
            end
            goto continue;
        end

        _G.SeasonFutureTable[data.serverType] = _G.SeasonFutureTable[data.serverType] or {}
        local insert = false;
        for idx, futureConf in ipairs(_G.SeasonFutureTable[data.serverType]) do
            if data.startTime < futureConf.startTime then
                insert = true;
                table.insert(_G.SeasonFutureTable[data.serverType], idx, data);
                break
            end
        end
        if not insert then
            table.insert(_G.SeasonFutureTable[data.serverType], data);
        end

        ::continue::
    end
    _G.SeasonTimeTable = seasonTimeTable

    _G.SeasonRankAwardGroupTable = {}
    for _, data in pairs(_G.SeasonRankAwardTable) do
        local rankAward = {}
        for _, info in pairs(data.rankAward) do
            local v = _G.SplitString(info, "|");
            table.insert(rankAward,{resId = tonumber(v[1]),count = tonumber(v[2])})
        end
        data.rankAward = rankAward

        local paidRankAward = {}
        for _, info in pairs(data.paidRankAward) do
            local v = _G.SplitString(info, "|");
            table.insert(paidRankAward, {resId = tonumber(v[1]),count = tonumber(v[2])})
        end
        data.paidRankAward = paidRankAward

        _G.SeasonRankAwardGroupTable[data.groupId] = _G.SeasonRankAwardGroupTable[data.groupId] or {}
        _G.SeasonRankAwardGroupTable[data.groupId][data.id] = data;
    end
end

function LoadSwitchTable()
    local env = dive.GetProperty("fabao.env", "ONLINE");

    local tmpLabelSwitchTable = {}
    for id, data in pairs(_G.LabelSwitchTable) do
        if env == "TEST" or env == "DEVELOP" then
            tmpLabelSwitchTable[id] = data.testValue;
        elseif env == "PRE" then
            tmpLabelSwitchTable[id] = data.preValue;
        elseif env == "ONLINE" then
            tmpLabelSwitchTable[id] = data.onlineValue;
        end
    end
    tmpLabelSwitchTable[1] = _G.versionAtoi(tmpLabelSwitchTable[1]);
    tmpLabelSwitchTable[2] = _G.versionAtoi(tmpLabelSwitchTable[2]);
    tmpLabelSwitchTable[3] = _G.versionAtoi(tmpLabelSwitchTable[3]);
    tmpLabelSwitchTable[4] = _G.versionAtoi(tmpLabelSwitchTable[4]);
    _G.LabelSwitchTable = tmpLabelSwitchTable

    local tmpServerConfigTable = {};
    for _, data in pairs(_G.ServerConfigTable) do
        if env == "TEST" or env == "DEVELOP" then
            tmpServerConfigTable[data.id] = data.testValue[1];
        elseif env == "PRE" then
            tmpServerConfigTable[data.id] = data.preValue[1];
        elseif env == "ONLINE" then
            tmpServerConfigTable[data.id] = data.onlineValue[1];
        end
    end
    _G.ServerConfigTable = tmpServerConfigTable;

    local tmpGamePlatformTable = {}
    for _, gamePlatformConfig in pairs(_G.GamePlatformTable) do
        tmpGamePlatformTable[gamePlatformConfig.platformStr] = gamePlatformConfig.matchStr
    end
    _G.GamePlatformTable = tmpGamePlatformTable
end

-- ActivityTable.xlsx
function LoadActivityTable()
    --所有配置合法的活动
    _G.activityTableAll = {};

    --当前时间区服有效的活动
    _G.activityTable = {};

    local activityAwardsTable = {};
    for id, data in pairs(_G.ActivityAwardTable) do
        activityAwardsTable[data.groupId] = activityAwardsTable[data.groupId] or {};

        local awards = {};
        local awardsClient = {};
        for _, award in pairs(data.awards) do
            local v = _G.SplitString(award, "|");
            if next(v) then
                awards[tonumber(v[1])] = tonumber(v[2]);
                table.insert(awardsClient, { resId = tonumber(v[1]), count = tonumber(v[2]) });
            end
        end

        data.awards = awards;
        data.awardsClient = awardsClient;

        -- condParam只能用数值, LobbyCommonLogic.proto的AwardStatusData为int64
        if data.condType == 1 then
            data.condParam = tonumber(data.condParam)
        else
            LogErr("ActivityAwardTable 奖励类型还未实现 id={} condType={}", id, data.condType)
        end
        data.desc = nil;

        -- 需要排序
        local hasInsert = false;
        for idx, award in ipairs(activityAwardsTable[data.groupId]) do
            if award.id > id then
                hasInsert = true
                table.insert(activityAwardsTable[data.groupId], idx, data);
                break
            end
        end
        if not hasInsert then
            table.insert(activityAwardsTable[data.groupId], data);
        end
    end

    local now = _G.GetCurTime();
    for id, data in pairs(_G.ActivityTable) do
        if data.startType == 0 or data.startType == 1 then
            if data.startTime and data.startTime ~= "" then
                data.startTime = math.ceil(_G.GetUnixTimeStamp(data.startTime));
            else
                data.startTime = 0
            end

            if data.endTime and data.endTime ~= "" then
                data.endTime = math.ceil(_G.GetUnixTimeStamp(data.endTime));
            else
                data.endTime = 0
            end

        elseif data.startType == 2 then
            data.startTime = tonumber(data.startTime);
            data.endTime = tonumber(data.endTime);
        end


        if data.dailyStartTime ~= "" then
            data.dailyStartTime = _G.GetUnixTimeSec(data.dailyStartTime);
        else
            data.dailyStartTime = 0;
        end
        if data.dailyEndTime ~= "" then
            data.dailyEndTime = _G.GetUnixTimeSec(data.dailyEndTime);
        else
            data.dailyEndTime = 0;
        end

        local activityAwards = activityAwardsTable[data.awardGroupId];
        if activityAwards then
            data.awardsConf = activityAwards;
        else
            data.awardsConf = {};
        end

        local awardsStepAwards = activityAwardsTable[data.awardStepGroupId];
        if awardsStepAwards then
            data.awardsStepConf = awardsStepAwards;
        else
            data.awardsStepConf = {};
        end

        _G.activityTableAll[id] = data;

        if data.endTime <= 0 or now < data.endTime or data.startType == 2 then
            _G.activityTable[id] = data;
        end

        local openServer = {};
        for _, serverType in pairs(data.serverType) do
            openServer[serverType] = 1;
        end
        data.serverType = openServer;
    end

    -- ActivityDailyFireSeed
    _G.DailyFireSeedTable = {};
    _G.DailyFireSeedClient = {};
    for _, data in pairs(_G.ActivityDayFireTable) do
        local rewards = {};
        local rewardsClient = {};
        for _, reward in pairs(data.awards) do
            local v = _G.SplitString(reward, "|");
            if next(v) then
                rewards[tonumber(v[1])] = tonumber(v[2]);
                table.insert(rewardsClient, { resId = tonumber(v[1]), count = tonumber(v[2]) });
            end
        end

        _G.DailyFireSeedTable[data.id] = {
            rewards = rewards,
            startTime = data.startTime,
            overTime = data.overTime
        }
        _G.DailyFireSeedClient[data.id] = {
            rewards = rewardsClient,
            startTime = data.startTime,
            overTime = data.overTime
        }
    end

    _G.WorldCompetitionTable = {};
    for id, data in pairs(_G.ActivityWorldCompetitionTable) do
        local rewards = {};
        for _, reward in pairs(data.items) do
            local v = _G.SplitString(reward, "|");
            if next(v) then
                table.insert(rewards, { resId = tonumber(v[1]), count = tonumber(v[2]) })
            end
        end
        _G.WorldCompetitionTable[id] = {
            rewards = rewards
        }
    end

    -- ActivityOnlineReward
    for _, data in pairs(_G.ActivityOnlineRewardTable) do
        local addList = {}
        for _, info in pairs(data.items) do
            local v = _G.SplitString(info, "|");
            table.insert(addList, { resId = tonumber(v[1]), count = tonumber(v[2]) })
        end
        data.items = addList
    end

    _G.ActivityTaskGroupTable = {}
    for _, data in pairs(_G.ActivityTaskTable) do
        if not _G.ActivityTaskGroupTable[data.groupId] then
            _G.ActivityTaskGroupTable[data.groupId] ={}
            _G.ActivityTaskGroupTable[data.groupId][data.id] = data;
        else
            _G.ActivityTaskGroupTable[data.groupId][data.id] = data;
        end
    end
end

function LoadFormationTable()
    for _, data in pairs(_G.FormationTable) do
        local conditions = {}
        for _, condition in pairs(data.conditions) do
            local cond = {}
            local v = _G.SplitString(condition, "|")
            for _, value in pairs(v) do
                table.insert(cond, tonumber(value))
            end
            table.insert(conditions, cond)
        end
        local formulas = {}
        for _, element in pairs(data.formula) do
            local v = _G.SplitString(element, "|")
            local elements = {}
            for _, value in pairs(v) do
                table.insert(elements, tonumber(value))
            end
            table.insert(formulas, elements)
        end
        data.conditions = conditions
        data.formulas = formulas
    end
    _G.ReactionMap = {}
    for _, data in pairs(_G.ReactionStateTable) do
        for idx, attr in pairs(data.baseAttr) do
            local formationAttr = data.baseAttr[idx]
            local sutraAttr = data.sutraAttr[idx]
            _G.ReactionMap[formationAttr] = _G.ReactionMap[formationAttr] or {}
            _G.ReactionMap[formationAttr][sutraAttr] = data
        end
    end
end

function LoadPveTable()
    -- 处理奖励字段
    for _, data in pairs(_G.PveMainDungeonLevelTable) do
        local rewards = {}
        for _, reward in pairs(data.rewardItems) do
            local v = _G.SplitString(reward, "|")
            table.insert(rewards, { resId = tonumber(v[1]), count = tonumber(v[2]) })
        end
        data.rewardItems = rewards

        local revenueRate = {}
        for _, reward in pairs(data.revenueRate) do
            local v = _G.SplitString(reward, "|")
            table.insert(revenueRate, { itemId = tonumber(v[1]), minCount = tonumber(v[2]), maxCount = tonumber(v[3]) })
        end
        data.revenueRate = revenueRate
    end

    _G.PveMainDungeonChapterRewardTable = _G.PveMDChapterRewardTable;
    -- 处理奖励字段
    for _, data in pairs(_G.PveMDChapterRewardTable) do
        local rewards = {}
        for _, reward in pairs(data.rewards) do
            local v = _G.SplitString(reward, "|")
            table.insert(rewards, { resId = tonumber(v[1]), count = tonumber(v[2]) })
        end
        data.rewards = rewards
    end

    -- 处理奖励字段
    for i, data in pairs(_G.PveNpcDungeonLevelTable) do
        local rewards = {}
        for _, reward in pairs(data.rewardItems) do
            local v = _G.SplitString(reward, "|")
            table.insert(rewards, { resId = tonumber(v[1]), count = tonumber(v[2]) })
        end
        data.rewardItems = rewards
    end

    for _, data in pairs(_G.GuildPveLevelTable) do
        data.stageRewards = {}
        for idx, value in ipairs(data.stageList) do
            local rewardGroupId = data.stageRewardList[idx]
            local rewardGroupConfig = (_G.RewardGroupTable or {})[rewardGroupId]
            local rewardsClient = {}
            if rewardGroupConfig then
                for _, reward in pairs(rewardGroupConfig.rewards or {}) do
                    local v = _G.SplitString(reward, "|");
                    if next(v) then
                        table.insert(rewardsClient, { resId = tonumber(v[1]), count = tonumber(v[2]) });
                    end
                end
            end
            table.insert(data.stageRewards, { killMonsterCount = value, rewards = rewardsClient })
        end
    end

    for _, data in pairs(_G.GuildBossLevelTable) do
        local memberAwards = {};
        for _, info in pairs(data.guildBossMemberRewards) do
            local v = _G.SplitString(info, "|");
            local rewardGroupConfig = (_G.RewardGroupTable or {})[tonumber(v[2])]
            local rewardsClient = {}
            if rewardGroupConfig then
                for _, reward in pairs(rewardGroupConfig.rewards or {}) do
                    local awardData = _G.SplitString(reward, "|");
                    if next(awardData) then
                        table.insert(rewardsClient, { resId = tonumber(awardData[1]), count = tonumber(awardData[2]) });
                    end
                end
            end

            table.insert( memberAwards, #memberAwards + 1, { cond = tonumber(v[1]),  rewards =  rewardsClient});
        end
        data.memberAwards = memberAwards;

        local rewardsClient = {}
        for _, reward in pairs(data.roundAwardItems) do
            local v = _G.SplitString(reward, "|");
            if next(v) then
                table.insert(rewardsClient, { resId = tonumber(v[1]), count = tonumber(v[2]) });
            end
        end
        data.rewardsClient = rewardsClient;
    end
end

function LoadRefinerTable()
    for _, data in pairs(_G.RefinerRealmTable) do
        local attrAddition = {}
        for _, addition in pairs(data.attrAddition) do
            local attr = _G.SplitString(addition, "|")
            table.insert(attrAddition, { attrId = tonumber(attr[1]), attrValue = tonumber(attr[2]) })
        end
        data.attrAddition = attrAddition
    end

    local tmpRefinerStageLevelTable = {};
    for _, data in pairs(_G.RefinerLevelTable) do
        local stage = data.realmStage
        local level = data.realmLevel

        tmpRefinerStageLevelTable[stage] = tmpRefinerStageLevelTable[stage] or {}
        tmpRefinerStageLevelTable[stage][level] = data
    end
    _G.RefinerStageLevelTable = tmpRefinerStageLevelTable

    for _, data in pairs(_G.RefinerFurnaceTable) do
        local costs = {}
        for _, cost in pairs(data.costs) do
            local v = _G.SplitString(cost, "|")
            table.insert(costs, { resId = tonumber(v[1]), count = tonumber(v[2]) })
        end

        local getItems = {}
        for _, getItem in pairs(data.getItems) do
            local v = _G.SplitString(getItem, "|")
            table.insert(getItems, { prob = tonumber(v[1]), itemId = tonumber(v[2]), count = tonumber(v[3]), limit = tonumber(v[4]), stack = tonumber(v[5]) })
        end

        data.costs = costs
        data.getItems = getItems
    end

    _G.RefinerTask2PoolId = {}
    for poolId, data in pairs(_G.RefinerCardPoolTable) do
        if data.openType == 1 then
            data.openParam = tonumber(data.openParam)
            _G.RefinerTask2PoolId[data.openParam] = poolId
        elseif data.openType == 2 then
            if data.openParam and data.openParam ~= "" then
                data.openParam = math.ceil(_G.GetUnixTimeStamp(data.openParam));
            else
                data.openParam = 0
            end
        elseif data.openType == 3 then
            data.openParam = tonumber(data.openParam) * 3600
        elseif data.openType == 4 then
            if data.openParam and data.openParam ~= "" then
                data.openParam = tonumber(data.openParam)
            else
                data.openParam = 0
            end
        else
            data.openParam = 0
        end

        local seasonIds = {}
        for _, seasonId in pairs(data.seasonIds) do
            seasonIds[seasonId] = 1;
        end
        data.seasonIds = seasonIds

        local specialCards = {}
        for _, cardId in pairs(data.specialCards) do
            specialCards[cardId] = cardId
        end
        data.specialCards = specialCards

        data.cardsKey = {}
        for _, cardId in pairs(data.cards) do
            data.cardsKey[cardId] = cardId
        end

        if data.openClientVersion and data.openClientVersion ~= "" then
            data.openClientVersion = _G.versionAtoi(data.openClientVersion);
        else
            data.openClientVersion = 0
        end
    end

end

function LoadShopTable()
    -- task开启的shop
    _G.TaskId2ShopIds = {};
    _G.ShopType2ShopIds = {};
    _G.TaskId2LimitTask = {};
    for shopId, data in pairs(_G.ShopTable) do
        if not data.openType or data.openType == 1 then
            data.openParam = 0
        elseif data.openType == 2 then
            data.openTaskIds = {}
            local taskIds = _G.SplitString(data.openParam, ";");
            for _, taskId in pairs(taskIds) do
                data.openTaskIds[tonumber(taskId)] = 1
                _G.TaskId2ShopIds[tonumber(taskId)] = _G.TaskId2ShopIds[tonumber(taskId)] or {}
                table.insert(_G.TaskId2ShopIds[tonumber(taskId)], shopId)
            end
        elseif data.openType == 3 then
            data.openParam = math.ceil(_G.GetUnixTimeStamp(data.openParam));
        else
            data.openParam = 0
        end

        if not data.closeType or data.closeType == 1 then
            data.closeParam = 0
        elseif data.closeType == 2 or data.closeType == 5 then
            data.closeParam = tonumber(data.closeParam)
        elseif data.closeType == 3 then
            data.closeParam = math.ceil(_G.GetUnixTimeStamp(data.closeParam));
        else
            data.closeParam = 0
        end

        _G.ShopType2ShopIds[data.shopType] = _G.ShopType2ShopIds[data.shopType] or {}
        table.insert(_G.ShopType2ShopIds[data.shopType], shopId)

        if data.shopType == 8 and data.limitTask > 0 then
            _G.TaskId2LimitTask[data.limitTask] = _G.TaskId2LimitTask[data.limitTask] or {}
            table.insert(_G.TaskId2LimitTask[data.limitTask], shopId)
        end

        if data.openClientVersion and data.openClientVersion ~= "" then
            data.openClientVersion = _G.versionAtoi(data.openClientVersion);
        else
            data.openClientVersion = 0
        end
    end

    -- 根据shopId的任务ID
    _G.TaskShopIdGoodsIds = {};

    --商品道具表
    -- 根据shopId的缓存
    _G.Shop2InGameGoodsTable = {};
    -- 免费商品
    _G.freeInGameGoods = {};
    for id, data in pairs(_G.InGameGoodsTable) do
        local items = {}
        for _, item in pairs(data.items) do
            table.insert(items, { resId = item[1], count = item[2] })
        end
        data.items = items

        _G.Shop2InGameGoodsTable[data.shopId] = _G.Shop2InGameGoodsTable[data.shopId] or {}
        _G.Shop2InGameGoodsTable[data.shopId][id] = data;

        if data.openType == 1 then
            local taskId = tonumber(data.openParam)
            _G.TaskShopIdGoodsIds[taskId] = _G.TaskShopIdGoodsIds[taskId] or {};
            _G.TaskShopIdGoodsIds[taskId][data.shopId] = _G.TaskShopIdGoodsIds[taskId][data.shopId] or {}
            _G.TaskShopIdGoodsIds[taskId][data.shopId][id] =  id
        end

        if data.price == 0 then
            _G.freeInGameGoods[data.shopId] = _G.freeInGameGoods[data.shopId] or {}
            _G.freeInGameGoods[data.shopId][id] = data;
        end

        if data.openClientVersion and data.openClientVersion ~= "" then
            data.openClientVersion = _G.versionAtoi(data.openClientVersion);
        else
            data.openClientVersion = 0
        end
    end

    -- 首充表
    -- 首充任务id2chargeId
    _G.TaskId2ChargeId = {}
    for id, data in pairs(_G.FirstChargeTable) do
        local todayItems = {};
        for _, todayItem in pairs(data.todayItems) do
            local v = _G.SplitString(todayItem, "|");
            if next(v) then
                table.insert(todayItems, { resId = tonumber(v[1]), count = tonumber(v[2]) });
            end
        end
        data.todayItems = todayItems;

        local nextDayItems = {};
        for _, nextDayItem in pairs(data.nextDayItems) do
            local v = _G.SplitString(nextDayItem, "|");
            if next(v) then
                table.insert(nextDayItems, { resId = tonumber(v[1]), count = tonumber(v[2]) });
            end
        end
        data.nextDayItems = nextDayItems;

        _G.TaskId2ChargeId[data.nextDayTask] = id
    end

    -- 基金奖励表
    local FundAwardGroupTable = {};
    for id, data in pairs(_G.AwardGroupTable) do
        local freeItems = {};
        for _, freeItem in pairs(data.freeItems) do
            local v = _G.SplitString(freeItem, "|");
            if next(v) then
                table.insert(freeItems, { resId = tonumber(v[1]), count = tonumber(v[2]) });
            end
        end
        data.freeItems = freeItems;

        local payItems = {};
        for _, payItem in pairs(data.payItems) do
            local v = _G.SplitString(payItem, "|");
            if next(v) then
                table.insert(payItems, { resId = tonumber(v[1]), count = tonumber(v[2]) });
            end
        end
        data.payItems = payItems;

        FundAwardGroupTable[data.groupId] = FundAwardGroupTable[data.groupId] or {}
        table.insert(FundAwardGroupTable[data.groupId], data)
        
        if data.conditionType == 1 then
            local taskConf = _G.TaskTable[data.conditionParam]
            if taskConf then
                data.conditionDesc = taskConf.name
            end
        end
    end
    for groudId, groudData in pairs(FundAwardGroupTable) do
        table.sort(groudData, function(a, b)
            return a.id < b.id
        end)
    end
    _G.FundAwardGroupTable = FundAwardGroupTable
    
    -- 基金任务id2fundData
    _G.TaskId2FundData = {}
    -- 基金表
    for id, data in pairs(_G.FundTable) do
        data.awardConfig = _G.FundAwardGroupTable[data.itemGroup]
        for awardId, awardConfig in pairs(data.awardConfig) do
            _G.TaskId2FundData[awardConfig.conditionParam] = {fundId=id, awardId=awardConfig.id}
        end
    end

    -- 卡数据
    for id, data in pairs(_G.ShopCardsTable) do
        local awards = {};
        for _, award in pairs(data.awards) do
            local v = _G.SplitString(award, "|");
            if next(v) then
                table.insert(awards, { resId = tonumber(v[1]), count = tonumber(v[2]) });
            end
        end
        data.awards = awards;

        local everyDayAwards = {};
        for _, award in pairs(data.everyDayAwards) do
            local v = _G.SplitString(award, "|");
            if next(v) then
                table.insert(everyDayAwards, { resId = tonumber(v[1]), count = tonumber(v[2]) });
            end
        end
        data.everyDayAwards = everyDayAwards;

        local serverRights = {};
        for _, right in pairs(data.rights) do
            serverRights[right] = 1;
        end
        data.serverRights = serverRights;
    end

    -- 商城id对应cards数据
    _G.Shop2ShopCardsTable = {}
    -- 卡片对应的shopId
    _G.CardId2ShopIdTable = {}
    -- 正在卖的首充数据
    _G.FirstChargeGoods = {};
    -- fundId 2 goodId
    _G.FundId2GoodIdTable = {};
    -- 根据shopId的缓存
    _G.Shop2GoodsTable = {};
    -- 根据shopId的限时礼包价格排序缓存
    _G.Shop2TimeLimitIdList = {};
    for id, data in pairs(_G.MonetaryGoodsTable) do
        local shopConfig = _G.ShopTable[data.shopId]
        if shopConfig.shopType == 8 then
            -- 限时礼包
            local items = {}
            for _, item in pairs(data.items) do
                table.insert(items, { resId = item[1], count = item[2] })
            end
            data.items = items

            _G.Shop2TimeLimitIdList[data.shopId] = _G.Shop2TimeLimitIdList[data.shopId] or {}
            local timeLimitIdList = _G.Shop2TimeLimitIdList[data.shopId]
            local insert = false;
            for idx, timeLimit in ipairs(timeLimitIdList) do
                if data.price < timeLimit.price then
                    insert = true;
                    table.insert(timeLimitIdList, idx, {price=data.price, goodsId=id});
                    break
                end
            end
            if not insert then
                table.insert(timeLimitIdList, {price=data.price, goodsId=id});
            end

        elseif shopConfig.shopType == 3 or shopConfig.shopType == 5 then
            -- 需要绑定权益卡表
            data.cardId = tonumber(data.right)
            data.items = nil;
            _G.Shop2ShopCardsTable[data.shopId] = _G.Shop2ShopCardsTable[data.shopId] or {}
            _G.Shop2ShopCardsTable[data.shopId][id] = _G.ShopCardsTable[data.cardId]
            _G.CardId2ShopIdTable[data.cardId] = data.shopId
        elseif shopConfig.shopType == 4 then
            -- 需要绑定基金表
            data.fundId = tonumber(data.right)
            data.items = nil;
            _G.FundId2GoodIdTable[data.fundId] = id
        elseif shopConfig.shopType == 7 then
            -- 需要绑定首充表
            data.firstChargeId = tonumber(data.right)
            data.items = nil;
            _G.FirstChargeGoods[id] = data
        else
            local items = {}
            for _, item in pairs(data.items) do
                table.insert(items, { resId = item[1], count = item[2] })
            end
            data.items = items
        end

        _G.Shop2GoodsTable[data.shopId] = _G.Shop2GoodsTable[data.shopId] or {}
        _G.Shop2GoodsTable[data.shopId][id] = data;

        if data.openType == 1 then
            local taskId = tonumber(data.openParam)
            _G.TaskShopIdGoodsIds[taskId] = _G.TaskShopIdGoodsIds[taskId] or {};
            _G.TaskShopIdGoodsIds[taskId][data.shopId] = _G.TaskShopIdGoodsIds[taskId][data.shopId] or {}
            _G.TaskShopIdGoodsIds[taskId][data.shopId][id] =  id
        end

        if data.openClientVersion and data.openClientVersion ~= "" then
            data.openClientVersion = _G.versionAtoi(data.openClientVersion);
        else
            data.openClientVersion = 0
        end
    end

    
end

---@class SutraTableData 法宝表数据

---@field baseCondition number 初始被动技能条件
---@field maxPassive number 满星被动技能id

    local tmpSutraLevelTable = {};
    for _, data in pairs(_G.SutraLevelTable) do
        local levelType = data.levelType
        local level = data.level
        tmpSutraLevelTable[levelType] = tmpSutraLevelTable[levelType] or {}
        tmpSutraLevelTable[levelType][level] = data
    end
    _G.SutraLevelTable = tmpSutraLevelTable

    local tmpSutraLevelAffixTable = {};
    for _, data in pairs(_G.SutraLevelAffixTable) do

        tmpSutraLevelAffixTable[data.levelAffixType][data.level] = data
    end

    local tmpSutraQualityTable = {}
    for _, data in pairs(_G.SutraQualityTable) do
        local sutraId = data.sutraId
        local quality = data.quality
        local params = {}
        for _, param in pairs(data.sacrificeParam) do
            local tmpSkills = {}
            local skills = _G.SplitString(param, "|")
            for _, skill in pairs(skills) do
                table.insert(tmpSkills, tonumber(skill))

        data.sacrificeParam = params

        tmpSutraQualityTable[sutraId] = tmpSutraQualityTable[sutraId] or {}
        tmpSutraQualityTable[sutraId][quality] = data
    end
    _G.SutraQualityTable = tmpSutraQualityTable

    local tmpSutraLevelDefTable = {};
    for _, data in pairs(_G.SutraLevelDefTable) do
        local level = data.level
        tmpSutraLevelDefTable[level] = data
    end
    _G.SutraLevelDefTable = tmpSutraLevelDefTable

    for _, data in pairs(_G.SutraQualityDefTable) do
        local breakDownItems = {}
        for _, info in pairs(data.breakDownItems) do
            local v = _G.SplitString(info, "|");
            table.insert(breakDownItems, { resId = tonumber(v[1]), count = tonumber(v[2]) })
        end
        data.breakDownItems = breakDownItems
    end
end

---@class BattleGameTypeTableData 玩法表数据
---@field id number 玩法id
---@field gameType number 玩法类型
---@field name string 玩法名称
---@field roundWaitTime number 回合等待时间
---@field gameWaitTime number 对局开始等待时间（动效）
---@field bossWaitTime number Boss预告时间
---@field mana number 法力自然获取速率
---@field keyUnlockRound table<number> 阵眼解锁回合
---@field monsterGroups table<number> 怪物组
---@field monsterLevel number 怪物等级
---@field sumMoney table<number> 刷牌消耗灵力
---@field defaultMoney number 初始灵力
---@field reward number 奖励配置id
---@field mapId number 地图id
---@field battleEndId number 战斗结束规则id
---@field preQueue table<number, table> 备选卡牌
---@field gameExtraTime number pve轮次时间
---@field battleData string 战斗快照数据
---@field roundEndType number 回合结束类型 1. 击杀boss 2. 轮次时间结束
function LoadBattleTable()
    _G.BattleRule = _G.BattleRuleTable[1]
    _G.MapConfig = _G.BattleMapConfigTable

    for _, data in pairs(_G.BattleGameTypeTable) do
        for teamId, queue in pairs(data.preQueue) do
            local newQue = {}
            for i = 1, #queue, 2 do
                local card = { id = queue[i], element = queue[i + 1] }
                table.insert(newQue, card)
            end
            data.preQueue[teamId] = newQue
        end

        for teamId, queue in pairs(data.preSpellQueue) do
            local newQue = {}
            for i = 1, #queue, 1 do
                local card = { id = queue[i], level = 1 }
                table.insert(newQue, card)
            end
            data.preSpellQueue[teamId] = newQue
        end
    end

    local tmpMonsterGroupRoundTable = {}
    for _, data in pairs(_G.BattleMonsterGroupTable) do
        local groupId = data.groupId
        local rounds = data.rounds
        for _, round in pairs(rounds) do
            tmpMonsterGroupRoundTable[round] = tmpMonsterGroupRoundTable[round] or {}
            tmpMonsterGroupRoundTable[round][groupId] = true
        end
    end
    _G.MonsterGroupRoundTable = tmpMonsterGroupRoundTable


    -- 战斗增益/减益
    for _, data in pairs(_G.BattleGainTable) do
        local tmpGainValue = {};
        for _, gainValue in pairs(data.gainValue) do
            local v = _G.SplitString(gainValue, "|");
            tmpGainValue[tonumber(v[1])] = tonumber(v[2])
        end
        data.gainValue = tmpGainValue
    end
end

function LoadServerTable()
    local serverTable = {}
    for serverId, serverConfig in pairs(_G.ServerTable) do
        serverConfig.serverStartTime = _G.GetUnixTimeStamp(serverConfig.serverStartTime);

        if serverConfig.beginFlyTime == "" then
            serverConfig.beginFlyTime = 0
        else
            serverConfig.beginFlyTime = _G.GetUnixTimeStamp(serverConfig.beginFlyTime);
        end

        if serverConfig.stopShowTime == "" then
            serverConfig.stopShowTime = 0
        else
            serverConfig.stopShowTime = _G.GetUnixTimeStamp(serverConfig.stopShowTime);
        end

        serverConfig.ascensionTarget = tostring(serverConfig.ascensionTarget)

        local flyTasks = {}
        for _, taskId in pairs(serverConfig.ascensionParam) do
            flyTasks[taskId] = 1
        end

function LoadMailTable()

        end
    end

function LoadItemTable()

        elseif dropClassConfig.dropType == _G.DROP_TYPE.INDEPENDENT then
            dropClassGroup.independentDrop[id] = dropClassConfig;
        end
        ::continue::
    end

    for _, data in pairs(_G.ItemTable) do

        for _, itemInfo in pairs(data.itemLists) do

-- 解析等级相关字段
-- {数组}|{数组}....
function parseLevelFiled(strings)
    local params = {}
    for _, str in pairs(strings) do
        local array = {}
        -- 删除string前后的{}，并根据;分割
        local len = string_len(str)
        if len > 1 then
            local tmp = string_sub(str, 2, len - 1)
            local tmpArray = _G.SplitString(tmp, ",")
            for _, v in pairs(tmpArray) do
                table.insert(array, tonumber(v))
            end
        end
        table.insert(params, array)
    end
    return params
end

        data.clientReward = rewardItems
        data.sweepClientAward = sweepAwards
    end

    for _, data in pairs(_G.SpiritPveChapterRewardTable) do
        local rewardItems = {}
        for _, info in pairs(data.rewards) do
            local v = _G.SplitString(info, "|");
            table.insert(rewardItems, { resId = tonumber(v[1]), count = tonumber(v[2]) });
        end

        data.rewards = rewardItems
    end
